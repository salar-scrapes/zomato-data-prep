#!/usr/bin/env python3
"""
audit_gold.py -- human audit of an LLM-generated gold set, BLIND to retrieval results

    python audit_gold.py make        # -> data/eval_work/gold_audit.csv  (sample of queries + the context needed to judge them)
    #  open in Excel/LibreOffice, fill answerable / ambiguity / copies_source_wording, save as CSV UTF-8
    python audit_gold.py summarize   # -> reports/gold_audit_summary.json  (aggregates only, safe to commit)

Rules
  * Do the audit BEFORE looking at any per-query evaluation result. Never edit or drop a query because of how it scored.
  * For each query judge only: can the gold outlet's profile/reviews answer it, would MANY other restaurants satisfy it
    equally well, and does it copy the wording of its source review.
  * The audit sheet contains dataset text (profile + review): keep it out of git (data/ is ignored); the summary has none.
"""
from __future__ import annotations

import argparse
import io
import json
import math
import sys
from pathlib import Path

import numpy as np
import pandas as pd

ALLOWED = {"answerable": {"y", "n"}, "ambiguity": {"low", "medium", "high"}, "copies_source_wording": {"y", "n"}}
AUDIT_COLS = ["query_id", "query_type", "query", "answerable", "ambiguity", "copies_source_wording", "notes",
              "gold_profile_text", "source_review_text"]


def sample_queries(gold: pd.DataFrame, n: int, seed: int = 0) -> pd.DataFrame:
    """Stratified (by query_type) deterministic sample of n queries; at least one per type when n allows."""
    rng = np.random.default_rng(seed)
    n = min(n, len(gold))
    counts = gold["query_type"].value_counts()
    alloc = {t: min(int(c), max(1, round(n * c / len(gold)))) for t, c in counts.items()}
    while sum(alloc.values()) > n:
        alloc[max(alloc, key=alloc.get)] -= 1
    while sum(alloc.values()) < n:
        room = [t for t in alloc if alloc[t] < counts[t]]
        if not room:
            break
        alloc[max(room, key=lambda t: counts[t] - alloc[t])] += 1
    picked: list[int] = []
    for t, k in alloc.items():
        pool = gold.index[gold["query_type"] == t].to_numpy()
        picked += rng.choice(pool, size=k, replace=False).tolist()
    return gold.loc[sorted(picked)]


def make_sheet(gold: pd.DataFrame, profiles: dict, review_text: dict, n: int = 20, seed: int = 0) -> pd.DataFrame:
    s = sample_queries(gold, n, seed)
    has_src = "source_review_id" in s.columns
    rows = []
    for r in s.itertuples(index=False):
        src = getattr(r, "source_review_id", None) if has_src else None
        rows.append({
            "query_id": r.query_id, "query_type": r.query_type, "query": r.query,
            "answerable": "", "ambiguity": "", "copies_source_wording": "", "notes": "",
            "gold_profile_text": " || ".join(profiles.get(k, "") for k in r.gold_outlet_keys),
            "source_review_text": review_text.get(src, "") if isinstance(src, str) else "",
        })
    return pd.DataFrame(rows, columns=AUDIT_COLS)


def read_sheet_csv(path: Path) -> pd.DataFrame:
    """Excel-proof reader: UTF-8 (BOM) first, then cp1252; comma, semicolon or tab separated; everything as text."""
    raw = Path(path).read_bytes()
    try:
        text = raw.decode("utf-8-sig")
    except UnicodeDecodeError:
        print("WARNING: sheet is not UTF-8 (Excel 'CSV' default); decoding as cp1252. Save as 'CSV UTF-8'.", file=sys.stderr)
        text = raw.decode("cp1252", errors="replace")
    for sep in (",", ";", "\t"):
        df = pd.read_csv(io.StringIO(text), sep=sep, dtype=str, keep_default_na=False)
        if "query_id" in df.columns:
            return df
    raise SystemExit("Could not find a 'query_id' column in the sheet (wrong file or separator).")


def wilson(k: int, n: int, z: float = 1.96) -> list | None:
    if n == 0:
        return None
    p, d = k / n, 1 + z * z / n
    centre, half = p + z * z / (2 * n), z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n))
    return [round((centre - half) / d, 4), round((centre + half) / d, 4)]


def summarize(sheet: pd.DataFrame) -> dict:
    df = sheet.copy()
    for c in ALLOWED:
        df[c] = df[c].astype(str).str.strip().str.lower()
    filled = (df[list(ALLOWED)] != "").any(axis=1)
    done = df[filled]
    if done.empty:
        raise SystemExit("No audited rows found: fill answerable / ambiguity / copies_source_wording first.")
    errors = []
    for r in done.itertuples(index=False):
        for c, allowed in ALLOWED.items():
            v = getattr(r, c)
            if v not in allowed:
                errors.append(f"{r.query_id}: {c}={v!r} (allowed: {sorted(allowed)})")
    if errors:
        raise SystemExit("Audit sheet problems:\n  " + "\n  ".join(errors))
    n = len(done)

    def rate(mask) -> dict:
        k = int(mask.sum())
        return {"count": k, "rate": round(k / n, 4), "wilson95": wilson(k, n)}

    flagged = done[(done["answerable"] == "n") | (done["ambiguity"] == "high") | (done["copies_source_wording"] == "y")]
    return {
        "n_audited": n, "n_in_sheet": int(len(df)),
        "not_answerable": rate(done["answerable"] == "n"),
        "ambiguity_high": rate(done["ambiguity"] == "high"),
        "ambiguity_medium": rate(done["ambiguity"] == "medium"),
        "copies_source_wording": rate(done["copies_source_wording"] == "y"),
        "any_flag": rate(df.loc[done.index].query_id.isin(flagged.query_id)),
        "flagged_query_ids": flagged["query_id"].tolist(),
        "by_type": {t: int(c) for t, c in done["query_type"].value_counts().items()},
        "note": "n is small: read the Wilson intervals, not the point rates.",
    }


def load_jsonl(path: Path) -> pd.DataFrame:
    return pd.DataFrame([json.loads(x) for x in Path(path).read_text(encoding="utf-8").splitlines() if x.strip()])


def parse_args(argv=None) -> argparse.Namespace:
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    sub = p.add_subparsers(dest="cmd", required=True)
    m = sub.add_parser("make")
    m.add_argument("--gold", type=Path, default=Path("eval/gold_queries.jsonl"))
    m.add_argument("--index-input", type=Path, default=Path("data/processed/index_input.parquet"))
    m.add_argument("--b-chunks", type=Path, default=Path("data/processed/chunks_B.parquet"))
    m.add_argument("--n", type=int, default=20)
    m.add_argument("--seed", type=int, default=0)
    m.add_argument("--out", type=Path, default=Path("data/eval_work/gold_audit.csv"))
    m.add_argument("--force", action="store_true", help="overwrite an existing sheet (destroys your answers!)")
    s = sub.add_parser("summarize")
    s.add_argument("--sheet", type=Path, default=Path("data/eval_work/gold_audit.csv"))
    s.add_argument("--out", type=Path, default=Path("reports/gold_audit_summary.json"))
    return p.parse_args(argv)


def main(argv=None) -> int:
    a = parse_args(argv)
    if a.cmd == "make":
        if a.out.exists() and not a.force:
            raise SystemExit(f"{a.out} exists; refusing to overwrite your audit answers (use --force).")
        gold = load_jsonl(a.gold)
        idx = pd.read_parquet(a.index_input, columns=["review_id", "text"])
        b = pd.read_parquet(a.b_chunks, columns=["outlet_key", "chunk_type", "text"])
        profiles = dict(zip(b.loc[b["chunk_type"] == "profile", "outlet_key"], b.loc[b["chunk_type"] == "profile", "text"]))
        need = set(gold.get("source_review_id", pd.Series(dtype=str)).dropna())
        review_text = dict(zip(idx.loc[idx["review_id"].isin(need), "review_id"], idx.loc[idx["review_id"].isin(need), "text"]))
        sheet = make_sheet(gold, profiles, review_text, a.n, a.seed)
        a.out.parent.mkdir(parents=True, exist_ok=True)
        sheet.to_csv(a.out, index=False, encoding="utf-8-sig")
        print(f"wrote {a.out} ({len(sheet)} queries). Fill the 3 judgement columns, save as CSV UTF-8, then run `summarize`.")
        return 0
    result = summarize(read_sheet_csv(a.sheet))
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(result, indent=2))
    print(json.dumps(result, indent=2))
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""
build_chunks.py -- Day 2, step 2

reviews_clean.parquet + Day-1 outlets.parquet  ->  index_input.parquet
                                               ->  chunks_A.parquet        (800-char windows, 120 overlap)
                                               ->  chunks_A_w400.parquet   (sensitivity, 60 overlap)
                                               ->  chunks_A_w1600.parquet  (sensitivity, 240 overlap)
                                               ->  chunks_B.parquet        (profile chunk + review chunks)
                                               ->  outlet_collisions.csv   (audit of identity collapses)

    python build_chunks.py --outlets data/processed/outlets.parquet

Rules baked in (do not change after you have seen eval results):
  * One shared input frame: cross-outlet duplicate reviews (n_outlets_same_text > 1) are dropped
    for BOTH strategies. Residual-mojibake rows stay in the corpus (they are excluded from GOLD, not from the index).
  * Sizes are measured with the embedding model's own tokenizer, never in characters.
  * Strategy B never exceeds the model limit; Strategy A may, and the overflow is measured and reported
    (silent truncation by the embedder is a real cost of naive chunking).

Outlet identity comes from outlet_key.build_outlet_id (same function prepare_reviews.py used);
it is NOT re-implemented here.

Requires: pandas, pyarrow, pandera, transformers (tokenizer only), and outlet_key.py on PYTHONPATH
"""

# ------------------------------------------------------------------ tokenizer wrapper
class HFCounter:
    """Thin wrapper so the chunking logic only needs count() and offsets() (and is testable without HF)."""

    def __init__(self, name: str):
        from transformers import AutoTokenizer

        tok = AutoTokenizer.from_pretrained(name, use_fast=True)
        if not tok.is_fast:
            raise SystemExit("Need a fast tokenizer (offset mapping is used for run-on sentences).")
        tok.model_max_length = int(1e9)  # we enforce limits ourselves; silences length warnings
        self.tok = tok

    def count(self, texts: list[str], batch: int = 10_000) -> np.ndarray:
        out = np.empty(len(texts), dtype="int64")
        for i in range(0, len(texts), batch):
            enc = self.tok(texts[i:i + batch], add_special_tokens=False,
                           return_attention_mask=False, return_token_type_ids=False)
            out[i:i + batch] = [len(x) for x in enc["input_ids"]]
        return out

    def offsets(self, text: str) -> list[tuple[int, int]]:
        return self.tok(text, add_special_tokens=False, return_offsets_mapping=True)["offset_mapping"]


# ------------------------------------------------------------------- small helpers
def short_hash(outlet_id: str) -> str:
    return hashlib.sha1(outlet_id.encode("utf-8")).hexdigest()[:12]


def _missing(v) -> bool:
    return v is None or v is pd.NA or (isinstance(v, float) and np.isnan(v))


def clean_str(v, max_items: int | None = None) -> str | None:
    if _missing(v):
        return None
    if isinstance(v, (list, tuple, np.ndarray)):
        items = [str(x).strip() for x in v if not _missing(x) and str(x).strip()]
        if max_items:
            items = items[:max_items]
        s = ", ".join(items)
    else:
        parts = [p.strip() for p in str(v).split(",") if p.strip()]
        s = ", ".join(parts[:max_items] if max_items else parts)
    return s or None


def to_float(v) -> float | None:
    """4.1 / '4.1/5' / '1,200' -> float ; 'NEW', '-', NaN -> None."""
    if _missing(v):
        return None
    if isinstance(v, (int, float, np.integer, np.floating)):
        return float(v)
    m = re.search(r"\d+(?:\.\d+)?", str(v).replace(",", ""))
    return float(m.group()) if m else None


def yes_no(v) -> str | None:
    if _missing(v):
        return None
    s = str(v).strip().lower()
    if s in {"yes", "true", "1", "y"}:
        return "yes"
    if s in {"no", "false", "0", "n"}:
        return "no"
    return None


# ------------------------------------------------------------------- outlets / profile
def resolve_columns(columns, overrides: list[str]) -> dict[str, str]:
    cols = set(columns)
    mapping: dict[str, str] = {}
    for logical, cands in CANDIDATES.items():
        hit = next((c for c in cands if c in cols), None)
        if hit:
            mapping[logical] = hit
    for item in overrides or []:
        logical, _, phys = item.partition("=")
        if logical not in CANDIDATES or phys not in cols:
            raise SystemExit(f"--map {item!r}: unknown logical field or missing column. "
                             f"Logical fields: {sorted(CANDIDATES)}")
        mapping[logical] = phys
    missing = REQUIRED - set(mapping)
    if missing:
        raise SystemExit(f"Outlets table lacks required fields {sorted(missing)}. "
                         f"Columns present: {sorted(cols)}. Use --map logical=physical.")
    print("outlet column mapping:", json.dumps(mapping), file=sys.stderr)
    return mapping


PROFILE_FIELDS = ("rest_type", "cuisines", "dish_liked", "cost", "rating", "votes", "online_order", "book_table")
COLLISION_LOG_COLS = ["outlet_key", "outlet", "kept", "deciding_rule", "votes", "completeness", "tiebreak", "address"]


def compute_outlet_ids(df: pd.DataFrame, name_col: str, address_col: str, id_fn=build_outlet_id) -> list[str]:
    """Day-1 rows -> outlet_id through the shared build_outlet_id.

    CONTRACT (as used in the run that worked): id_fn(df, [name_col, address_col]) -> Series/list of str,
    one per row, same order as df.
    """
    for c in (name_col, address_col):
        if c not in df.columns:
            raise SystemExit(f"Outlets table has no column {c!r} (needed for build_outlet_id). "
                             f"Columns: {sorted(df.columns)}")
    bad = df[name_col].isna() | df[address_col].isna()
    if bad.any():
        raise SystemExit(f"{int(bad.sum())} outlets have a null name/address; cannot build outlet_id.")
    ids = list(id_fn(df, [name_col, address_col]))
    if len(ids) != len(df) or not all(isinstance(i, str) and i for i in ids):
        raise SystemExit("build_outlet_id must return one non-empty string per row -- signature mismatch?")
    return ids


def collapse_identity_collisions(df: pd.DataFrame, ids: list[str], cols: dict[str, str], address_col: str,
                                 tiebreak_col: str = "url") -> tuple[pd.DataFrame, list[str], pd.DataFrame]:
    """Day 1 collapsed on url; Day 2 identity is the normalised (name, address). Rows that now share an id
    are the same outlet listed under several urls -> keep exactly one, deterministically:

        most votes (the Day-1 rule)  ->  most complete profile  ->  smallest `tiebreak_col`  ->  file order

    Returns (kept_df, kept_ids, audit_log). Every decision is logged; nothing is dropped silently.
    """
    work = pd.DataFrame({"outlet_id": ids}, index=df.index)
    if not work["outlet_id"].duplicated().any():
        return df, ids, pd.DataFrame(columns=COLLISION_LOG_COLS)

    votes = [to_float(v) for v in df[cols["votes"]]] if "votes" in cols else [None] * len(df)
    work["_votes"] = [-1.0 if v is None else v for v in votes]  # unknown votes lose to any known votes
    present = [cols[f] for f in PROFILE_FIELDS if f in cols]
    work["_complete"] = [sum(clean_str(v) is not None for v in row)
                         for row in df[present].itertuples(index=False, name=None)]
    work["_tb"] = df[tiebreak_col].astype(str).to_numpy() if tiebreak_col in df.columns else ""
    work["_pos"] = np.arange(len(df))

    ranked = work.sort_values(["outlet_id", "_votes", "_complete", "_tb", "_pos"],
                              ascending=[True, False, False, True, True], kind="mergesort")
    ranked["_rank"] = ranked.groupby("outlet_id", sort=False).cumcount()

    dup_ids = work.loc[work["outlet_id"].duplicated(keep=False), "outlet_id"].unique()
    log = []
    for oid, g in ranked[ranked["outlet_id"].isin(dup_ids)].groupby("outlet_id", sort=False):
        win, runner = g.iloc[0], g.iloc[1]
        if win["_votes"] != runner["_votes"]:
            rule = "votes"
        elif win["_complete"] != runner["_complete"]:
            rule = "completeness"
        elif win["_tb"] != runner["_tb"]:
            rule = f"tiebreak:{tiebreak_col}"
        else:
            rule = "file_order"
        for rank, (_, r) in enumerate(g.iterrows()):
            log.append({
                "outlet_key": short_hash(oid), "outlet": oid.replace("\x1f", " | "), "kept": rank == 0,
                "deciding_rule": rule, "votes": None if r["_votes"] < 0 else r["_votes"],
                "completeness": int(r["_complete"]), "tiebreak": r["_tb"],
                "address": str(df.iloc[int(r["_pos"])][address_col]),
            })

    keep_pos = set(ranked.loc[ranked["_rank"] == 0, "_pos"])
    mask = work["_pos"].isin(keep_pos).to_numpy()
    return df.loc[mask].reset_index(drop=True), [i for i, m in zip(ids, mask) if m], pd.DataFrame(log)


def prep_outlets(df: pd.DataFrame, cols: dict[str, str], address_col: str = "address",
                 id_fn=build_outlet_id, tiebreak_col: str = "url",
                 max_collapse_share: float = 0.005) -> tuple[pd.DataFrame, pd.DataFrame]:
    """-> (outlets, collision_log). One row per composite outlet_id."""
    ids = compute_outlet_ids(df, cols["name"], address_col, id_fn)
    kept, ids, log = collapse_identity_collisions(df, ids, cols, address_col, tiebreak_col)
    dropped = len(df) - len(kept)
    if dropped / len(df) > max_collapse_share:
        raise SystemExit(f"Identity collapse would drop {dropped} of {len(df)} rows "
                         f"(> {max_collapse_share:.2%}). That is not 'a few typo variants' -- check build_outlet_id.")

    out = pd.DataFrame({"outlet_id": ids})
    if not out["outlet_id"].is_unique:  # cannot happen after the collapse; guard stays
        raise SystemExit("outlet_id still not unique after identity collapse.")
    out["outlet_key"] = [short_hash(x) for x in out["outlet_id"]]
    if not out["outlet_key"].is_unique:
        raise SystemExit("short_hash collision on outlet_key (extend the hash length).")
    for logical, phys in cols.items():
        out[logical] = pd.Series(kept[phys].to_numpy(dtype=object), index=out.index, dtype=object)
    out["rating_num"] = [to_float(v) for v in out["rating"]] if "rating" in out else np.nan
    out["cost_num"] = [to_float(v) for v in out["cost"]] if "cost" in out else np.nan
    return out, log


def profile_text(r: dict, max_dish: int = 15) -> str:
    """Templated, deterministic profile text. Numbers are kept coarse; filtering is not what is being tested."""
    head = clean_str(r.get("name")) or ""
    if (kind := clean_str(r.get("rest_type"))):
        head += f" - {kind}"
    if (loc := clean_str(r.get("location"))):
        head += f" in {loc}"
    parts = [head + "."]
    if (c := clean_str(r.get("cuisines"))):
        parts.append(f"Cuisines: {c}.")
    if (d := clean_str(r.get("dish_liked"), max_items=max_dish)):
        parts.append(f"Known for: {d}.")
    if (cost := to_float(r.get("cost"))) is not None:
        parts.append(f"Cost for two: about {int(cost)} rupees.")
    if (rt := to_float(r.get("rating"))) is not None:
        votes = to_float(r.get("votes"))
        parts.append(f"Rated {rt:.1f} out of 5" + (f" by {int(votes)} people." if votes is not None else "."))
    if (o := yes_no(r.get("online_order"))):
        parts.append(f"Online ordering: {o}.")
    if (b := yes_no(r.get("book_table"))):
        parts.append(f"Table booking: {b}.")
    return " ".join(parts)


# ------------------------------------------------------------------- shared input frame
def build_index_input(reviews: pd.DataFrame, outlets: pd.DataFrame) -> tuple[pd.DataFrame, dict]:
    rep = {"reviews_in": len(reviews)}
    keep = reviews[reviews["n_outlets_same_text"] == 1]
    rep["dropped_cross_outlet_spam"] = len(reviews) - len(keep)
    keep = keep.merge(outlets[["outlet_id", "outlet_key"]], on="outlet_id", how="left", validate="m:1")
    if keep["outlet_key"].isna().any():
        miss = keep.loc[keep["outlet_key"].isna(), "outlet_id"].unique()
        raise SystemExit(
            f"{len(miss)} outlet_ids in reviews do not exist in the outlets table built via build_outlet_id.\n"
            f"  review ids : {[repr(x) for x in miss[:3]]}\n"
            f"  outlets ids: {[repr(x) for x in outlets['outlet_id'].head(3)]}\n"
            f"Compare the two: different normalisation/version of build_outlet_id, or stale reviews_clean.parquet."
        )
    idx = keep[["review_id", "outlet_id", "outlet_key", "review_order", "text", "rating",
                "n_chars", "mojibake_residual"]].reset_index(drop=True)
    rep.update(reviews_indexed=len(idx), outlets_with_reviews=int(idx["outlet_id"].nunique()),
               outlets_total=len(outlets))
    return idx, rep


# ------------------------------------------------------------------ Strategy A (naive)
def window_spans(n: int, size: int, overlap: int) -> list[tuple[int, int]]:
    """Fixed character windows. Last window ends exactly at n (no tail-only duplicate chunk)."""
    if size <= overlap:
        raise ValueError("size must exceed overlap")
    spans, start = [], 0
    while n > 0:
        end = min(start + size, n)
        spans.append((start, end))
        if end == n:
            break
        start += size - overlap
    return spans


def finalize(df: pd.DataFrame, counter, max_content: int) -> pd.DataFrame:
    df = df.reset_index(drop=True)
    df["n_chars"] = df["text"].str.len().astype("int64")
    df["n_tokens"] = counter.count(df["text"].tolist())
    df["exceeds_model_limit"] = df["n_tokens"] > max_content
    return df


def build_chunks_A(idx: pd.DataFrame, size: int, overlap: int, counter, max_content: int) -> pd.DataFrame:
    """Per outlet: concatenate reviews in union order, then cut fixed windows. Blind to review boundaries."""
    blobs = (
        idx.sort_values(["outlet_id", "review_order"], kind="stable")
        .groupby("outlet_id", sort=False)
        .agg(outlet_key=("outlet_key", "first"), blob=("text", " ".join))
    )
    tag = f"A{size}"
    rows = []
    for oid, okey, blob in zip(blobs.index, blobs["outlet_key"], blobs["blob"]):
        for n, (a, b) in enumerate(window_spans(len(blob), size, overlap)):
            t = blob[a:b].strip()
            if t:
                rows.append((f"{okey}:w{size}:{n}", tag, oid, okey, "window", None, n, t))
    return finalize(pd.DataFrame(rows, columns=CHUNK_COLS), counter, max_content)


# --------------------------------------------------------------- Strategy B (profile + review)
def split_long_review(text: str, counter, target: int) -> list[tuple[int, int]]:
    """Sentence-aware split of ONE over-long review -> char spans, each <= `target` tokens.

    Greedy sentence packing with a one-sentence overlap (only when that sentence is short).
    A single run-on 'sentence' longer than `target` is cut by token offsets with HARD_SPLIT_OVERLAP.
    """
    cuts = [0] + [m.end() for m in SENT_SPLIT.finditer(text)] + [len(text)]
    sents = [(a, b) for a, b in zip(cuts, cuts[1:]) if b > a]
    toks = counter.count([text[a:b] for a, b in sents])

    units: list[tuple[int, int, int]] = []  # (start, end, n_tokens), every unit <= target
    for (a, b), t in zip(sents, toks):
        if t <= target:
            units.append((a, b, int(t)))
            continue
        offs = counter.offsets(text[a:b])
        n, w, stride = len(offs), 0, max(target - HARD_SPLIT_OVERLAP, target // 2, 1)
        while True:
            e = min(w + target, n)
            # The tokenizer's normaliser silently drops some characters (zero-width joiners, variation selectors,
            # U+FFFD, control chars), so token offsets can skip them. Anchor the first window to the sentence
            # start and the last to the sentence end; middle windows overlap, so their gaps are covered too.
            ua = a if w == 0 else a + offs[w][0]
            ub = b if e == n else a + offs[e - 1][1]
            units.append((ua, ub, e - w))
            if e == n:
                break
            w += stride

    spans, i, k = [], 0, len(units)
    while i < k:
        j, total = i, 0
        while j < k and total + units[j][2] <= target:
            total += units[j][2]
            j += 1
        spans.append((units[i][0], units[j - 1][1]))
        if j >= k:
            break
        i = j - 1 if (j - 1 > i and units[j - 1][2] <= target // 4) else j
    return spans


def uncovered_nonspace(text: str, spans: list[tuple[int, int]]) -> list[int]:
    """Positions of non-whitespace characters of `text` that lie in no span (empty list == lossless split)."""
    covered = np.zeros(len(text), dtype=bool)
    for a, b in spans:
        covered[a:b] = True
    nonspace = np.fromiter((not ch.isspace() for ch in text), dtype=bool, count=len(text))
    return np.flatnonzero(nonspace & ~covered).tolist()


def covers_all_nonspace(text: str, spans: list[tuple[int, int]]) -> bool:
    return not uncovered_nonspace(text, spans)


def build_chunks_B(idx: pd.DataFrame, outlets: pd.DataFrame, counter, max_content: int,
                   target: int, max_dish: int = 15) -> tuple[pd.DataFrame, dict]:
    # 1) one profile chunk per outlet (ALL outlets, including those with no reviews)
    prof = [
        (f"{r['outlet_key']}:p", "B", r["outlet_id"], r["outlet_key"], "profile", None, 0,
         profile_text(r, max_dish))
        for r in outlets.to_dict("records")
    ]

    # 2) one chunk per review; only reviews over the model limit are split
    n_tok = counter.count(idx["text"].tolist())
    fits = idx[n_tok <= max_content]
    long = idx[n_tok > max_content]
    rev = list(zip(
        (fits["review_id"] + ":0").tolist(), ["B"] * len(fits), fits["outlet_id"], fits["outlet_key"],
        ["review"] * len(fits), fits["review_id"], [0] * len(fits), fits["text"],
    ))
    n_split_chunks = 0
    for rid, oid, okey, text in zip(long["review_id"], long["outlet_id"], long["outlet_key"], long["text"]):
        spans = split_long_review(text, counter, target)
        lost = uncovered_nonspace(text, spans)
        if lost:
            detail = [(i, f"U+{ord(text[i]):04X}", unicodedata.name(text[i], "?")) for i in lost[:10]]
            raise SystemExit(f"Lossy split on review {rid} ({len(text)} chars, {len(spans)} spans): "
                             f"{len(lost)} non-space chars uncovered, first 10 (pos, codepoint, name): {detail}. "
                             f"Splitter bug -- refusing to continue.")
        for n, (a, b) in enumerate(spans):
            rev.append((f"{rid}:{n}", "B", oid, okey, "review", rid, n, text[a:b].strip()))
        n_split_chunks += len(spans)

    df = pd.DataFrame(prof + rev, columns=CHUNK_COLS)
    meta = outlets[["outlet_id", "location", "cuisines", "rating_num", "cost_num"]].rename(
        columns={"rating_num": "rating", "cost_num": "cost_for_two"})
    meta["location"] = meta["location"].map(clean_str)
    meta["cuisines"] = meta["cuisines"].map(clean_str)
    df = df.merge(meta, on="outlet_id", how="left", validate="m:1")
    df["rating"] = df["rating"].astype("float64")
    df["cost_for_two"] = df["cost_for_two"].astype("float64")
    rep = {"reviews_split": len(long), "chunks_from_split_reviews": n_split_chunks}
    return finalize(df, counter, max_content), rep

def _pa():
    try:
        import pandera.pandas as pa
    except ImportError:  # pandera < 0.24
        import pandera as pa
    return pa


def validate_chunks(df: pd.DataFrame, kind: str, expected_strategy: str, max_content: int,
                    failures_path: Path) -> pd.DataFrame:
    pa = _pa()
    cols = {
        "chunk_id": pa.Column(str, pa.Check.str_length(min_value=1), unique=True),
        "strategy": pa.Column(str, pa.Check.equal_to(expected_strategy)),
        "outlet_id": pa.Column(str, pa.Check.str_length(min_value=1)),
        "outlet_key": pa.Column(str, pa.Check.str_matches(r"^[0-9a-f]{12}$")),
        "chunk_type": pa.Column(str, pa.Check.isin(["window"] if kind == "A" else ["profile", "review"])),
        "review_id": pa.Column(str, nullable=True),
        "chunk_idx": pa.Column(int, pa.Check.ge(0)),
        "text": pa.Column(str, checks=[
            pa.Check(lambda s: s.str.len() > 0, error="empty text"),
            pa.Check(lambda s: s == s.str.strip(), error="untrimmed text"),
        ]),
        "n_chars": pa.Column(int, pa.Check.ge(1)),
        "n_tokens": pa.Column(int, pa.Check.ge(1)),
        "exceeds_model_limit": pa.Column(bool),
    }
    dfchecks = [
        pa.Check(lambda d: bool((d["exceeds_model_limit"] == (d["n_tokens"] > max_content)).all()),
                 error="exceeds_model_limit inconsistent with n_tokens"),
    ]
    if kind == "A":
        dfchecks.append(pa.Check(lambda d: bool(d["review_id"].isna().all()), error="A must carry no review_id"))
    else:
        cols["n_tokens"] = pa.Column(int, [pa.Check.ge(1), pa.Check.le(max_content)])  # B: hard limit
        cols["location"] = pa.Column(str, nullable=True)
        cols["cuisines"] = pa.Column(str, nullable=True)
        cols["rating"] = pa.Column(float, pa.Check.in_range(0.0, 5.0), nullable=True)
        cols["cost_for_two"] = pa.Column(float, pa.Check.ge(0), nullable=True)
        dfchecks += [
            pa.Check(lambda d: bool(((d["chunk_type"] == "review") == d["review_id"].notna()).all()),
                     error="review_id must be set exactly on review chunks"),
            pa.Check(lambda d: bool(d.loc[d["chunk_type"] == "profile", "outlet_key"].is_unique),
                     error="more than one profile chunk per outlet"),
        ]
    schema = pa.DataFrameSchema(cols, checks=dfchecks, strict=True, coerce=True)
    try:
        return schema.validate(df, lazy=True)
    except pa.errors.SchemaErrors as exc:
        exc.failure_cases.to_csv(failures_path, index=False)
        raise SystemExit(f"Pandera FAILED for {expected_strategy}: {len(exc.failure_cases)} failure cases "
                         f"-> {failures_path}\n"
                         f"{exc.failure_cases.groupby(['column', 'check']).size().to_string()}")


def validate_index_input(df: pd.DataFrame, failures_path: Path) -> pd.DataFrame:
    pa = _pa()
    schema = pa.DataFrameSchema(
        {
            "review_id": pa.Column(str, unique=True),
            "outlet_id": pa.Column(str),
            "outlet_key": pa.Column(str, pa.Check.str_matches(r"^[0-9a-f]{12}$")),
            "review_order": pa.Column(int, pa.Check.ge(0)),
            "text": pa.Column(str, pa.Check.str_length(min_value=1)),
            "rating": pa.Column(float, pa.Check.in_range(0.0, 5.0), nullable=True),
            "n_chars": pa.Column(int, pa.Check.ge(1)),
            "mojibake_residual": pa.Column(bool),
        },
        unique=["outlet_id", "review_order"], strict=True, coerce=True,
    )
    try:
        return schema.validate(df, lazy=True)
    except pa.errors.SchemaErrors as exc:
        exc.failure_cases.to_csv(failures_path, index=False)
        raise SystemExit(f"Pandera FAILED for index_input -> {failures_path}")


def frame_stats(df: pd.DataFrame) -> dict:
    q = df["n_tokens"].quantile([0.5, 0.9, 0.99]).to_dict()
    return {
        "chunks": len(df),
        "outlets": int(df["outlet_id"].nunique()),
        "chunks_per_outlet_mean": round(len(df) / max(df["outlet_id"].nunique(), 1), 2),
        "n_tokens_p50": int(q[0.5]), "n_tokens_p90": int(q[0.9]), "n_tokens_p99": int(q[0.99]),
        "n_tokens_max": int(df["n_tokens"].max()),
        "exceeds_model_limit": int(df["exceeds_model_limit"].sum()),
        "exceeds_model_limit_share": round(float(df["exceeds_model_limit"].mean()), 5),
    }

def main(argv=None) -> int:
    a = parse_args(argv)
    a.out_dir.mkdir(parents=True, exist_ok=True)
    max_content = a.max_seq_len - 2
    target = max_content - SPLIT_HEADROOM
    counter = HFCounter(a.tokenizer)

    reviews = pd.read_parquet(a.reviews)
    outlets_raw = pd.read_parquet(a.outlets)
    cols = resolve_columns(outlets_raw.columns, a.map)
    outlets, collisions = prep_outlets(outlets_raw, cols, a.address_col, tiebreak_col=a.tiebreak_col,
                                       max_collapse_share=a.max_collapse_share)
    if len(collisions):
        collisions.to_csv(a.out_dir / "outlet_collisions.csv", index=False, encoding="utf-8-sig")

    idx, report = build_index_input(reviews, outlets)
    report["identity"] = {
        "day1_rows": len(outlets_raw), "outlets_after_identity_collapse": len(outlets),
        "rows_collapsed": len(outlets_raw) - len(outlets),
        "collision_groups": int(collisions["outlet_key"].nunique()) if len(collisions) else 0,
        "deciding_rules": collisions.loc[collisions["kept"], "deciding_rule"].value_counts().to_dict()
        if len(collisions) else {},
    }
    idx = validate_index_input(idx, a.out_dir / "index_input_failures.csv")
    idx.to_parquet(a.out_dir / "index_input.parquet", index=False, compression="zstd")
    report["limits"] = {"max_seq_len": a.max_seq_len, "max_content_tokens": max_content, "split_target": target}

    # Strategy A: primary + sensitivity runs
    for size, overlap in A_VARIANTS.items():
        tag = f"A{size}"
        df = build_chunks_A(idx, size, overlap, counter, max_content)
        df = validate_chunks(df, "A", tag, max_content, a.out_dir / f"chunks_{tag}_failures.csv")
        name = "chunks_A.parquet" if size == PRIMARY_A else f"chunks_A_w{size}.parquet"
        df.to_parquet(a.out_dir / name, index=False, compression="zstd")
        report[tag] = {"file": name, "window": size, "overlap": overlap, **frame_stats(df)}

    # Strategy B
    B, b_rep = build_chunks_B(idx, outlets, counter, max_content, target, a.max_dish)
    B = validate_chunks(B, "B", "B", max_content, a.out_dir / "chunks_B_failures.csv")

    # Lineage invariants Pandera cannot express
    rev = B[B["chunk_type"] == "review"]
    if set(rev["review_id"]) != set(idx["review_id"]):
        raise SystemExit("Strategy B lost or invented reviews relative to index_input.")
    if int((B["chunk_type"] == "profile").sum()) != len(outlets):
        raise SystemExit("Strategy B profile chunk count != number of outlets.")

    B.to_parquet(a.out_dir / "chunks_B.parquet", index=False, compression="zstd")
    report["B"] = {"file": "chunks_B.parquet", **b_rep, **frame_stats(B),
                   "profile_chunks": len(outlets), "review_chunks": len(rev)}

    (a.out_dir / "chunks_report.json").write_text(json.dumps(report, indent=2))
    print(json.dumps(report, indent=2))
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""Outlet-level (cluster) bootstrap and sign-flip test for comparing two systems.

Why: several gold queries target the same outlet, so queries are not independent.
Resampling whole outlets gives honest confidence intervals for the paired difference.

Input: a per-query CSV (default reports/per_query.csv), one row per (system, query):
    query_id, system, outlet_key, hit1, hit5, rr
  hit1 / hit5 are 0/1 and rr is the reciprocal rank (0 on a miss). They must come
  from the same code that produced eval_report.json, so the means reproduce it.
  outlet_key is the query's gold outlet (the cluster).

Output: a table on screen and reports/cluster_bootstrap.json.
  - naive CI:   resamples queries (assumes independence; shown for comparison)
  - cluster CI: resamples whole outlets (the honest one)
  - sign-flip p: flips the sign of each outlet's total difference; exact when there
                 are <= 16 outlets, Monte Carlo otherwise

Usage:
    python cluster_bootstrap.py
    python cluster_bootstrap.py --treat "B|exact" --base "A800|exact" --n-boot 20000
"""
import argparse
import itertools
import json
import os
import sys

import numpy as np
import pandas as pd

METRICS = {"Hit@1": "hit1", "Hit@5": "hit5", "MRR": "rr"}
REQUIRED = ["query_id", "system", "outlet_key", "hit1", "hit5", "rr"]
EXACT_MAX_CLUSTERS = 16
MC_SIGN_DRAWS = 200_000


def load_pair(path, treat, base):
    df = pd.read_csv(path, dtype={"query_id": str, "system": str, "outlet_key": str})
    missing = [c for c in REQUIRED if c not in df.columns]
    if missing:
        sys.exit(f"{path} is missing columns {missing}; expected {REQUIRED}")
    systems = sorted(df["system"].unique())
    for s in (treat, base):
        if s not in systems:
            sys.exit(f"system {s!r} not found in {path}. Available: {systems}")

    parts = {}
    for name in (treat, base):
        sub = df[df["system"] == name]
        if sub["query_id"].duplicated().any():
            sys.exit(f"duplicate query_id rows for system {name!r}")
        parts[name] = sub.set_index("query_id")
    t, b = parts[treat], parts[base]
    if set(t.index) != set(b.index):
        sys.exit("the two systems cover different query_ids; a paired comparison needs identical sets")
    qids = sorted(t.index)
    t, b = t.loc[qids], b.loc[qids]
    if not (t["outlet_key"] == b["outlet_key"]).all():
        sys.exit("outlet_key differs between the systems for the same query_id")
    return t, b


def percentile_ci(x):
    lo, hi = np.percentile(x, [2.5, 97.5])
    return [float(lo), float(hi)]


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--file", default="reports/per_query.csv")
    ap.add_argument("--treat", default="B|exact", help="system being tested")
    ap.add_argument("--base", default="A800|exact", help="baseline system")
    ap.add_argument("--n-boot", type=int, default=10000)
    ap.add_argument("--seed", type=int, default=42)
    ap.add_argument("--out", default="reports/cluster_bootstrap.json")
    args = ap.parse_args()

    t, b = load_pair(args.file, args.treat, args.base)
    n = len(t)
    codes, outlets = pd.factorize(t["outlet_key"])
    c = len(outlets)
    if c < 2:
        sys.exit("need at least 2 outlets for a cluster bootstrap")
    counts = np.bincount(codes, minlength=c).astype(float)

    rng = np.random.default_rng(args.seed)
    cluster_idx = rng.integers(0, c, size=(args.n_boot, c))   # same draws for every metric
    query_idx = rng.integers(0, n, size=(args.n_boot, n))

    if c <= EXACT_MAX_CLUSTERS:
        signs = np.array(list(itertools.product((-1.0, 1.0), repeat=c)))
        perm_kind = f"exact, {len(signs)} sign patterns"
    else:
        signs = rng.choice((-1.0, 1.0), size=(MC_SIGN_DRAWS, c))
        perm_kind = f"Monte Carlo, {MC_SIGN_DRAWS} sign draws"

    results = {}
    for label, col in METRICS.items():
        tv, bv = t[col].to_numpy(float), b[col].to_numpy(float)
        d = tv - bv
        observed = float(d.mean())
        sums = np.bincount(codes, weights=d, minlength=c)  # per-outlet total difference

        naive = d[query_idx].mean(axis=1)
        cluster = sums[cluster_idx].sum(axis=1) / counts[cluster_idx].sum(axis=1)
        flipped = (signs @ sums) / n
        p = float(np.mean(np.abs(flipped) >= abs(observed) - 1e-12))

        results[label] = {
            "treat_mean": float(tv.mean()),
            "base_mean": float(bv.mean()),
            "diff": observed,
            "naive_ci": percentile_ci(naive),
            "cluster_ci": percentile_ci(cluster),
            "signflip_p": p,
        }

    print(f"{args.treat}  minus  {args.base}   |   {n} queries, {c} outlets, "
          f"{args.n_boot} resamples, seed {args.seed}")
    print(f"{'metric':<7} {'treat':>6} {'base':>6} {'diff':>7}   {'naive 95% CI':<18} "
          f"{'outlet-cluster 95% CI':<24} {'sign-flip p':>11}")
    for label, r in results.items():
        nlo, nhi = r["naive_ci"]
        clo, chi = r["cluster_ci"]
        verdict = "excludes 0" if (clo > 0 or chi < 0) else "includes 0"
        print(f"{label:<7} {r['treat_mean']:>6.3f} {r['base_mean']:>6.3f} {r['diff']:>+7.3f}   "
              f"[{nlo:+.3f}, {nhi:+.3f}]    [{clo:+.3f}, {chi:+.3f}] {verdict:<10} {r['signflip_p']:>8.4f}")
    print(f"sign-flip test: {perm_kind}")
    print("Check: treat/base means should match eval_report.json to 3 decimals.")

    payload = {
        "treat": args.treat, "base": args.base,
        "n_queries": n, "n_outlets": c,
        "n_boot": args.n_boot, "seed": args.seed,
        "signflip": perm_kind, "metrics": results,
    }
    os.makedirs(os.path.dirname(args.out) or ".", exist_ok=True)
    with open(args.out, "w", encoding="utf-8") as f:
        json.dump(payload, f, indent=2)
    print(f"Saved {args.out}")


if __name__ == "__main__":
    main()

def _norm(v) -> str:
    return WS.sub(" ", str(v)).strip().lower()


def diagnose(df: pd.DataFrame, name: str, address: str, url: str | None = None,
             votes: str | None = None, expect: int = 12_499) -> dict:
    out: dict = {"rows": len(df)}
    print(f"rows                         : {len(df):,}   (Day-1 outlet table should be ~{expect:,})")
    print(f"columns                      : {list(df.columns)}")
    for c in (name, address):
        if c not in df.columns:
            raise SystemExit(f"column {c!r} not in file")
    print(f"null name / null address     : {int(df[name].isna().sum())} / {int(df[address].isna().sum())}")

    raw_pairs = len(df[[name, address]].astype(str).drop_duplicates())
    ids = df[name].map(_norm) + "\x1f" + df[address].map(_norm)
    out.update(unique_raw_pairs=raw_pairs, unique_norm_ids=int(ids.nunique()))
    print(f"unique (name,address) raw    : {raw_pairs:,}")
    print(f"unique normalised ids        : {ids.nunique():,}")
    if url and url in df.columns:
        out["unique_urls"] = int(df[url].nunique())
        print(f"unique {url:<22}: {df[url].nunique():,}")

    dup = ids.duplicated(keep=False)
    n_groups = int(ids[dup].nunique())
    out.update(dup_rows=int(dup.sum()), dup_groups=n_groups)
    print(f"rows sharing an id           : {int(dup.sum()):,}  in {n_groups:,} groups")
    if not dup.any():
        print("-> ids ARE unique with plain normalisation; the problem is how build_outlet_id is being called.")
        return out

    d = df[dup].assign(_id=ids[dup]).copy()
    # which columns differ inside duplicate groups? (list/array cells compared as text)
    share = {}
    for c in df.columns:
        s = d[c].astype(str)
        share[c] = float((s.groupby(d["_id"]).nunique() > 1).mean())
    differing = {c: round(v, 3) for c, v in sorted(share.items(), key=lambda kv: -kv[1]) if v > 0}
    out["differing_columns_share"] = differing
    print("\nshare of duplicate groups in which each column DIFFERS (0 = identical copies):")
    for c, v in differing.items():
        print(f"  {c:<32}{v:6.1%}")
    if not differing:
        print("  (none: the duplicate rows are exact copies)")

    print("\nfirst 3 duplicate groups:")
    show = [c for c in (name, address, url, votes) if c and c in df.columns]
    for gid in list(d["_id"].drop_duplicates())[:3]:
        print(d.loc[d["_id"] == gid, show].to_string(index=False), "\n")
    return out

#!/usr/bin/env python3
"""
embed_chunks.py -- Phase 3, step 1

chunks_*.parquet  ->  <out-dir>/<variant>.npy            float32, L2-normalised, row order == chunk table
                      <out-dir>/<variant>.manifest.json  everything needed to trust / reproduce the vectors

    python embed_chunks.py --variants B A800 --bench 2000      # measure speed + ETA, writes nothing
    python embed_chunks.py --variants B A800                    # resumable: Ctrl+C / crash / reboot is safe
    python embed_chunks.py --variants all --device cuda         # on a free GPU (Kaggle / Colab)

Design rules
  * Source of truth = the .npy + manifest. Chroma (next step) is rebuilt from them, never the other way round.
  * Passages are embedded WITHOUT a prefix; queries use QUERY_PREFIX (recorded in the manifest so evaluate.py
    reads it from there -- one definition for A and B).
  * Work is sharded; a shard is written atomically; resuming refuses to mix shards from a different model
    revision / source file / row order / shard size.
  * Gates before the final file is written: right shape + float32, no NaN/inf, unit norms, and a re-embed
    self-check on random rows. Chunks longer than the model limit are counted (the embedder truncates silently).

Requires: pandas, pyarrow, numpy, torch, sentence-transformers
"""
from __future__ import annotations

import argparse
import hashlib
import json
import math
import os
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

import numpy as np
import pandas as pd

MODEL = "BAAI/bge-small-en-v1.5"
PASSAGE_PREFIX = ""
QUERY_PREFIX = "Represent this sentence for searching relevant passages: "
VARIANT_FILES = {
    "B": "chunks_B.parquet",
    "A800": "chunks_A.parquet",
    "A400": "chunks_A_w400.parquet",
    "A1600": "chunks_A_w1600.parquet",
}
IDENTITY_KEYS = ["model", "model_commit_hash", "max_seq_length", "passage_prefix", "normalize", "dtype",
                 "dim", "source_sha256", "ids_sha256", "shard_size", "rows"]
SELFCHECK_MIN_COS = 0.999


# ------------------------------------------------------------------------- embedder
class STEmbedder:
    """sentence-transformers wrapper. Anything with .encode/.dim/.max_seq_length/.device/.info works (tests)."""

    def __init__(self, model_name: str, revision: str | None, device: str | None,
                 batch_size: int | None, threads: int | None):
        import sentence_transformers
        import torch
        from sentence_transformers import SentenceTransformer

        if threads:
            torch.set_num_threads(threads)
        self.device = device or ("cuda" if torch.cuda.is_available() else "cpu")
        self.model = SentenceTransformer(model_name, device=self.device, revision=revision)
        self.max_seq_length = int(self.model.max_seq_length)
        if self.max_seq_length < 512:
            raise SystemExit(f"max_seq_length={self.max_seq_length} < 512: B chunks (up to 510 tokens) would be "
                             f"silently truncated. Refusing.")
        self.dim = int(self.model.get_sentence_embedding_dimension())
        self.batch_size = batch_size or (256 if self.device == "cuda" else 64)
        commit = getattr(getattr(self.model[0].auto_model, "config", None), "_commit_hash", None)
        self.info = {
            "model": model_name, "model_commit_hash": commit, "max_seq_length": self.max_seq_length,
            "dim": self.dim, "device": self.device, "batch_size": self.batch_size,
            "torch": torch.__version__, "sentence_transformers": sentence_transformers.__version__,
            "torch_threads": torch.get_num_threads(),
        }

    def encode(self, texts: list[str]) -> np.ndarray:
        return self.model.encode(texts, batch_size=self.batch_size, normalize_embeddings=True,
                                 convert_to_numpy=True, show_progress_bar=False).astype("float32")


# --------------------------------------------------------------------------- helpers
def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for block in iter(lambda: f.read(1 << 20), b""):
            h.update(block)
    return h.hexdigest()


def sha256_ids(ids: list[str]) -> str:
    return hashlib.sha256("\n".join(ids).encode("utf-8")).hexdigest()


def check_embeddings(E: np.ndarray, rows: int, dim: int) -> None:
    if E.shape != (rows, dim) or E.dtype != np.float32:
        raise SystemExit(f"Embedding matrix has shape/dtype {E.shape}/{E.dtype}, expected {(rows, dim)}/float32.")
    if not np.isfinite(E).all():
        raise SystemExit("Embeddings contain NaN/inf.")
    norms = np.linalg.norm(E, axis=1)
    if (np.abs(norms - 1.0) > 1e-3).any():
        raise SystemExit(f"Embeddings are not unit-norm (min {norms.min():.5f}, max {norms.max():.5f}).")


def _atomic_save(path: Path, arr: np.ndarray) -> None:
    tmp = path.with_name(path.stem + ".tmp.npy")
    np.save(tmp, arr)
    os.replace(tmp, path)


def effective_tokens(df: pd.DataFrame, max_seq_length: int) -> np.ndarray:
    """Tokens the model actually reads: content + [CLS]/[SEP], capped at the model limit."""
    return np.minimum(df["n_tokens"].to_numpy(dtype="int64") + 2, max_seq_length)


def _fmt_eta(seconds: float) -> str:
    h, rem = divmod(int(seconds), 3600)
    return f"{h}h{rem // 60:02d}m"


# ------------------------------------------------------------------------ core logic
def embed_table(df: pd.DataFrame, emb, out_dir: Path, name: str, source_sha: str, shard_size: int = 5000,
                selfcheck_rows: int = 200, force: bool = False, log=print) -> dict:
    """Embed one chunk table (columns: chunk_id, text, n_tokens) -> <name>.npy + manifest. Resumable."""
    out_dir.mkdir(parents=True, exist_ok=True)
    ids = df["chunk_id"].tolist()
    texts = df["text"].tolist()
    n = len(df)
    if n == 0 or len(set(ids)) != n:
        raise SystemExit(f"[{name}] chunk table empty or chunk_id not unique.")
    if any((not isinstance(t, str)) or not t.strip() for t in texts):
        raise SystemExit(f"[{name}] empty/non-string text in chunk table.")

    identity = {
        "model": emb.info["model"], "model_commit_hash": emb.info.get("model_commit_hash"),
        "max_seq_length": emb.max_seq_length, "passage_prefix": PASSAGE_PREFIX, "normalize": True,
        "dtype": "float32", "dim": emb.dim, "source_sha256": source_sha, "ids_sha256": sha256_ids(ids),
        "shard_size": shard_size, "rows": n,
    }
    final_path, manifest_path = out_dir / f"{name}.npy", out_dir / f"{name}.manifest.json"

    if manifest_path.exists() and final_path.exists() and not force:
        old = json.loads(manifest_path.read_text())
        if all(old.get(k) == identity[k] for k in IDENTITY_KEYS):
            log(f"[{name}] already complete and identical -> skipping")
            return old
        raise SystemExit(f"[{name}] existing output was built from a different identity "
                         f"(model/source/rows changed). Re-run with --force to rebuild.")

    shard_dir = out_dir / f"{name}.shards"
    shard_dir.mkdir(exist_ok=True)
    meta_path = shard_dir / "meta.json"
    devices = [emb.device]
    if meta_path.exists():
        old = json.loads(meta_path.read_text())
        diff = {k: (old.get(k), identity[k]) for k in IDENTITY_KEYS if old.get(k) != identity[k]}
        if diff:
            raise SystemExit(f"[{name}] refusing to resume: run identity changed {diff}. "
                             f"Delete {shard_dir} to restart from scratch.")
        devices = sorted(set(old.get("devices", [])) | {emb.device})
    meta_path.write_text(json.dumps({**identity, "devices": devices}, indent=2))

    n_shards = math.ceil(n / shard_size)
    eff = effective_tokens(df, emb.max_seq_length)
    todo = [k for k in range(n_shards) if not (shard_dir / f"{k:05d}.npy").exists()]
    todo_tokens = int(sum(eff[k * shard_size:(k + 1) * shard_size].sum() for k in todo))
    log(f"[{name}] {n:,} chunks, {n_shards} shards ({n_shards - len(todo)} already done), "
        f"{todo_tokens / 1e6:.1f}M tokens to embed on {emb.device}")

    t0, done_tokens = time.time(), 0
    for i, k in enumerate(todo):
        lo, hi = k * shard_size, min(n, (k + 1) * shard_size)
        E = emb.encode(texts[lo:hi])
        if E.shape != (hi - lo, emb.dim) or E.dtype != np.float32:
            raise SystemExit(f"[{name}] shard {k}: bad shape/dtype {E.shape}/{E.dtype}")
        _atomic_save(shard_dir / f"{k:05d}.npy", E)
        done_tokens += int(eff[lo:hi].sum())
        el = time.time() - t0
        tps = done_tokens / max(el, 1e-9)
        log(f"[{name}] shard {k + 1}/{n_shards} | {(hi - lo) * (i + 1) / max(el, 1e-9):.1f} chunks/s (avg) | "
            f"{tps:,.0f} tok/s | ETA {_fmt_eta((todo_tokens - done_tokens) / max(tps, 1e-9))}")

    E = np.concatenate([np.load(shard_dir / f"{k:05d}.npy") for k in range(n_shards)]).astype("float32", copy=False)
    check_embeddings(E, n, emb.dim)

    # self-check: re-embed random rows, they must reproduce the stored vectors
    rng = np.random.default_rng(0)
    sel = np.sort(rng.choice(n, size=min(selfcheck_rows, n), replace=False))
    E2 = emb.encode([texts[i] for i in sel])
    cos = (E[sel] * E2).sum(axis=1)
    if cos.min() < SELFCHECK_MIN_COS:
        raise SystemExit(f"[{name}] self-check FAILED: min cosine {cos.min():.5f} < {SELFCHECK_MIN_COS} "
                         f"between stored and re-embedded rows (non-deterministic or mixed-model shards).")

    _atomic_save(final_path, E)
    manifest = {
        **identity, "variant": name, "devices": devices, "model_info": emb.info,
        "query_prefix": QUERY_PREFIX, "n_truncated_at_embedding": int((df["n_tokens"] + 2 > emb.max_seq_length).sum()),
        "selfcheck_min_cosine": float(cos.min()), "selfcheck_rows": int(len(sel)),
        "norm_min": float(np.linalg.norm(E, axis=1).min()), "norm_max": float(np.linalg.norm(E, axis=1).max()),
        "created_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
    }
    manifest_path.write_text(json.dumps(manifest, indent=2))
    log(f"[{name}] DONE -> {final_path.name} | truncated at embedding: {manifest['n_truncated_at_embedding']} "
        f"| self-check min cosine {cos.min():.6f}")
    return manifest


def bench_table(df: pd.DataFrame, emb, name: str, n: int, log=print) -> dict:
    """Embed a random sample, report throughput and a full-variant ETA. Writes nothing."""
    sample = df.sample(min(n, len(df)), random_state=0)
    t = time.time()
    emb.encode(sample["text"].tolist())
    dt = time.time() - t
    tok_s = effective_tokens(sample, emb.max_seq_length).sum() / dt
    total = effective_tokens(df, emb.max_seq_length).sum()
    res = {"variant": name, "chunks_per_s": len(sample) / dt, "tokens_per_s": float(tok_s),
           "total_tokens_M": total / 1e6, "eta_s": float(total / tok_s)}
    log(f"[{name}] bench: {res['chunks_per_s']:.1f} chunks/s | {tok_s:,.0f} tok/s | "
        f"{res['total_tokens_M']:.1f}M tokens total -> full-variant ETA {_fmt_eta(res['eta_s'])}")
    return res


# ----------------------------------------------------------------------------------- CLI
def parse_args(argv=None) -> argparse.Namespace:
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument("--chunks-dir", type=Path, default=Path("data/processed"))
    p.add_argument("--out-dir", type=Path, default=Path("data/embeddings"))
    p.add_argument("--variants", nargs="+", default=["B", "A800"], choices=[*VARIANT_FILES, "all"],
                   help="default: the primary comparison only (B, A800)")
    p.add_argument("--model", default=MODEL)
    p.add_argument("--revision", default=None, help="pin a HF commit hash (taken from an earlier manifest)")
    p.add_argument("--device", default=None, help="cpu | cuda (default: auto)")
    p.add_argument("--batch-size", type=int, default=None, help="default 64 on CPU, 256 on GPU")
    p.add_argument("--threads", type=int, default=None, help="torch CPU threads (default: torch's choice)")
    p.add_argument("--shard-size", type=int, default=5000)
    p.add_argument("--bench", type=int, default=0, metavar="N", help="benchmark N random chunks per variant, write nothing")
    p.add_argument("--force", action="store_true", help="rebuild even if a complete output exists")
    return p.parse_args(argv)


def main(argv=None) -> int:
    a = parse_args(argv)
    variants = list(VARIANT_FILES) if "all" in a.variants else a.variants
    emb = STEmbedder(a.model, a.revision, a.device, a.batch_size, a.threads)
    print(json.dumps(emb.info), file=sys.stderr)

    for v in variants:
        path = a.chunks_dir / VARIANT_FILES[v]
        if not path.exists():
            raise SystemExit(f"{path} not found")
        df = pd.read_parquet(path, columns=["chunk_id", "text", "n_tokens"])

            bench_table(df, emb, v, a.bench)
        else:
            embed_table(df, emb, a.out_dir, v, sha256_file(path), a.shard_size, force=a.force)
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""
evaluate.py -- Phase 4: Strategy A vs Strategy B, scored at OUTLET level (Hit@k, MRR)

    python evaluate.py --gold eval/gold_queries.jsonl --variants B A800

Gold file (JSON lines, one query per line):
    {"query_id": "q001", "query": "quiet place for a date night, slow service though",
     "query_type": "review_evidence",                 # review_evidence | profile_fact | mixed
     "gold_outlet_keys": ["a1b2c3d4e5f6"],            # outlet_key(s) from chunks_*.parquet / index_input.parquet; NEVER url
     "source_review_id": "0123456789abcdef"}          # optional: review the query was written from

What this enforces (aborts with the full problem list if the gold file breaks a rule):
  * every gold outlet has indexed reviews (otherwise Strategy A cannot retrieve it: unfair by construction)
  * source reviews exist in the shared input frame and are not residual-mojibake
  * unique ids, known query types, no duplicate queries, at least --min-queries queries
What it measures:
  * PRIMARY (fixed in advance): B vs A800, exact engine, ΔHit@5, cluster-bootstrap 95% CI. Everything else is exploratory.
  * exact numpy search is the oracle; Chroma (if its collection exists) is scored the same way and the two engines are
    compared; Chroma never decides a verdict.
  * control: B_restricted = B with outlets that have no indexed reviews removed from results (A cannot return them).
  * leakage: queries sharing a --leak-ngram-word run with a gold outlet's text are flagged and re-scored without them.
  * uncertainty: bootstrap over OUTLET clusters (queries about the same outlet are not independent), exact McNemar on
    Hit@5 discordant pairs, per-type breakdown (CI only for n >= 10).
Requires: pandas, numpy, pyarrow, sentence-transformers (query embedding), chromadb (optional engine)
"""

# ------------------------------------------------------------------------- gold file
def load_gold(path: Path) -> pd.DataFrame:
    rows = [json.loads(line) for line in path.read_text(encoding="utf-8").splitlines() if line.strip()]
    return pd.DataFrame(rows)


def validate_gold(gold: pd.DataFrame, outlets_with_reviews: set, reviews: pd.DataFrame | None = None,
                  min_queries: int = 30) -> list[str]:
    """-> list of human-readable problems (empty list == gold set is acceptable)."""
    need = {"query_id", "query", "query_type", "gold_outlet_keys"}
    if need - set(gold.columns):
        return [f"missing columns: {sorted(need - set(gold.columns))}"]
    probs: list[str] = []
    if len(gold) < min_queries:
        probs.append(f"only {len(gold)} queries (< {min_queries}): differences would be noise")
    for qid in gold["query_id"][gold["query_id"].duplicated()].unique():
        probs.append(f"duplicate query_id {qid!r}")
    norm = gold["query"].astype(str).str.lower().str.replace(r"\W+", " ", regex=True).str.strip()
    for qid in gold.loc[norm.duplicated(keep=False), "query_id"]:
        probs.append(f"{qid}: duplicate query text")
    has_src = "source_review_id" in gold.columns
    for r in gold.itertuples(index=False):
        qid = r.query_id
        if not isinstance(r.query, str) or len(r.query.split()) < 3:
            probs.append(f"{qid}: query is empty or shorter than 3 words")
        if r.query_type not in QUERY_TYPES:
            probs.append(f"{qid}: query_type {r.query_type!r} not in {QUERY_TYPES}")
        keys = r.gold_outlet_keys
        if not isinstance(keys, (list, tuple)) or not keys or len(set(keys)) != len(keys):
            probs.append(f"{qid}: gold_outlet_keys must be a non-empty list without duplicates")
            continue
        for k in keys:
            if not isinstance(k, str) or not HEX12.match(k):
                probs.append(f"{qid}: gold key {k!r} is not a 12-hex outlet_key (url / outlet_id are not allowed)")
            elif k not in outlets_with_reviews:
                probs.append(f"{qid}: gold outlet {k} has no indexed reviews (Strategy A cannot retrieve it) -> relabel")
        src = getattr(r, "source_review_id", None) if has_src else None
        if isinstance(src, str) and src and reviews is not None:
            if src not in reviews.index:
                probs.append(f"{qid}: source_review_id {src} not in the shared input frame (dropped as spam, or typo)")
            else:
                row = reviews.loc[src]
                if row["outlet_key"] not in keys:
                    probs.append(f"{qid}: source review belongs to {row['outlet_key']}, not a gold outlet")
                if bool(row["mojibake_residual"]):
                    probs.append(f"{qid}: source review is residual-mojibake (excluded from gold by rule)")
    return probs


def _words(s: str) -> list[str]:
    return re.findall(r"[a-z0-9']+", s.lower())


def _ngrams(words: list[str], n: int) -> set:
    return {tuple(words[i:i + n]) for i in range(len(words) - n + 1)}


def leaky_flags(gold: pd.DataFrame, corpus_by_outlet: dict, n: int = 5) -> np.ndarray:
    """True where the query shares an n-word run with any text of a gold outlet (copy-paste / near copy)."""
    cache: dict = {}
    flags = []
    for q, keys in zip(gold["query"], gold["gold_outlet_keys"]):
        grams = _ngrams(_words(q), n)
        leak = False
        for k in keys:
            if k not in cache:
                cache[k] = set().union(*[_ngrams(_words(t), n) for t in corpus_by_outlet.get(k, [])]) \
                    if corpus_by_outlet.get(k) else set()
            if grams & cache[k]:
                leak = True
                break
        flags.append(leak)
    return np.array(flags, dtype=bool)


# ---------------------------------------------------------------------------- scoring
def rank_outlets(rows, outlet_of, allowed=None) -> tuple[list, list]:
    """Chunk rows in rank order -> (unique outlets in rank order, the best-ranked chunk row of each)."""
    seen, outs, best = set(), [], []
    for r in rows:
        o = outlet_of[r]
        if o in seen or (allowed is not None and o not in allowed):
            continue
        seen.add(o)
        outs.append(o)
        best.append(int(r))
    return outs, best


def score_lists(row_lists, outlet_of, gold_sets, allowed=None, depth: int | None = None) -> dict:
    m = len(gold_sets)
    s = {k: np.zeros(m) for k in METRICS}
    s.update(n_outlets=np.zeros(m, dtype=int), rank=np.full(m, np.nan),
             first_gold_row=np.full(m, -1, dtype=int), top1_row=np.full(m, -1, dtype=int))
    for q, rows in enumerate(row_lists):
        rows = list(rows)[:depth] if depth else list(rows)
        outs, best = rank_outlets(rows, outlet_of, allowed)
        s["n_outlets"][q] = len(outs)
        if best:
            s["top1_row"][q] = best[0]
        r = next((i + 1 for i, o in enumerate(outs) if o in gold_sets[q]), None)
        if r is not None:
            s["rank"][q], s["first_gold_row"][q] = r, best[r - 1]
            s["hit1"][q], s["hit5"][q], s["hit10"][q] = r == 1, r <= 5, r <= 10
            s["rr"][q], s["rr10"][q] = 1.0 / r, (1.0 / r if r <= 10 else 0.0)
    return s


# ----------------------------------------------------------------------- statistics
def make_weights(cluster_ids, n_boot: int = 10_000, seed: int = 0) -> np.ndarray:
    """Cluster-bootstrap weights (n_boot, n_queries): resample OUTLETS with replacement; every query of a drawn
    outlet comes along. Same weights for every metric/system, so paired differences stay paired."""
    uniq, inv = np.unique(np.asarray(cluster_ids), return_inverse=True)
    nc, rng = len(uniq), np.random.default_rng(seed)
    counts = np.stack([np.bincount(rng.integers(0, nc, nc), minlength=nc) for _ in range(n_boot)])
    return counts[:, inv].astype(np.float64)


def mean_ci(W: np.ndarray, x, mask=None, with_ci: bool = True) -> dict:
    x = np.asarray(x, dtype=float)
    if mask is not None:
        W, x = W[:, mask], x[mask]
    n = len(x)
    if n == 0:
        return {"mean": None, "ci": None, "n": 0}
    out = {"mean": float(x.mean()), "ci": None, "n": int(n)}
    if with_ci:
        den = W.sum(axis=1)
        ok = den > 0
        lo, hi = np.percentile((W[ok] @ x) / den[ok], [2.5, 97.5])
        out["ci"] = [float(lo), float(hi)]
    return out


def mcnemar_exact_p(b: int, c: int) -> float:
    """Two-sided exact McNemar (binomial) p-value from the discordant counts."""
    n = b + c
    if n == 0:
        return 1.0
    return min(1.0, 2 * sum(math.comb(n, i) for i in range(min(b, c) + 1)) / 2 ** n)


def summarize(s: dict, W: np.ndarray, mask=None, with_ci: bool = True) -> dict:
    m = np.ones(len(s["hit5"]), bool) if mask is None else mask
    out = {name: mean_ci(W, s[key], m, with_ci) for key, name in METRICS.items()}
    out["short_list_rate_lt5_outlets"] = float((s["n_outlets"][m] < 5).mean()) if m.any() else None
    out["outlets_returned_mean"] = float(s["n_outlets"][m].mean()) if m.any() else None
    return out


def compare(sa: dict, sb: dict, W: np.ndarray, mask=None) -> dict:
    """b minus a (a = baseline)."""
    m = np.ones(len(sa["hit5"]), bool) if mask is None else mask
    ha, hb, ra, rb = sa["hit5"][m], sb["hit5"][m], sa["rr"][m], sb["rr"][m]
    b_only, a_only = int(((hb == 1) & (ha == 0)).sum()), int(((ha == 1) & (hb == 0)).sum())
    return {
        "n": int(m.sum()),
        "hit5_diff": mean_ci(W, sb["hit5"] - sa["hit5"], m),
        "mrr_diff": mean_ci(W, sb["rr"] - sa["rr"], m),
        "hit5_discordant": {"b_only": b_only, "a_only": a_only, "mcnemar_exact_p": mcnemar_exact_p(b_only, a_only)},
        "rr_wins_losses_ties": {"b_better": int((rb > ra).sum()), "a_better": int((ra > rb).sum()),
                                "tie": int((ra == rb).sum())},
    }


def verdict(diff: dict, a: str = "A800", b: str = "B") -> str:
    if diff["ci"] is None:
        return "no data"
    lo, hi = diff["ci"]
    if lo > 0:
        return f"{b} > {a}: 95% CI excludes 0"
    if hi < 0:
        return f"{a} > {b}: 95% CI excludes 0"
    return "inconclusive: 95% CI includes 0"


# ----------------------------------------------------------------------- core (pure)
def evaluate(variants: dict, Q: np.ndarray, gold: pd.DataFrame, leaky: np.ndarray, outlets_with_reviews: set,
             chroma_rows: dict | None = None, k_fetch: int = 150, sens_depths=(50, 100), n_boot: int = 10_000,
             seed: int = 0) -> tuple[dict, pd.DataFrame]:
    gold_sets = [set(k) for k in gold["gold_outlet_keys"]]
    qtype = gold["query_type"].to_numpy()
    W = make_weights([k[0] for k in gold["gold_outlet_keys"]], n_boot, seed)

    systems: dict = {}  # name -> dict(scores, rows, variant, allowed)
    for vn, v in variants.items():
        ex = exact_topk(v.E, Q, k_fetch)[0]
        systems[f"{vn}|exact"] = {"rows": ex, "v": v, "allowed": None}
        if chroma_rows and vn in chroma_rows:
            systems[f"{vn}|chroma"] = {"rows": chroma_rows[vn], "v": v, "allowed": None}
        if vn == "B" and "A800" in variants:
            systems["B_restricted|exact"] = {"rows": ex, "v": v, "allowed": outlets_with_reviews}
        if vn == "B":  # control: review chunks only
            keep = np.flatnonzero(v.chunk_type == "review")
            systems["B_reviews_only|exact"] = {"rows": keep[exact_topk(v.E[keep], Q, k_fetch)[0]], "v": v, "allowed": None}
    for s in systems.values():
        s["scores"] = score_lists(s["rows"], s["v"].outlet_of, gold_sets, s["allowed"])

    report: dict = {
        "meta": {"n_queries": len(gold), "n_outlet_clusters": int(len({k[0] for k in gold["gold_outlet_keys"]})),
                 "query_type_counts": {t: int((qtype == t).sum()) for t in QUERY_TYPES},
                 "leaky_queries": int(leaky.sum()), "k_fetch_chunks": k_fetch, "n_boot": n_boot, "seed": seed,
                 "systems": list(systems)},
        "systems": {n: summarize(s["scores"], W) for n, s in systems.items()},
        "by_type": {}, "exploratory": {}, "engine_agreement": {},
    }
    for t in QUERY_TYPES:
        mt = qtype == t
        if mt.any():
            report["by_type"][t] = {n: summarize(s["scores"], W, mt, with_ci=bool(mt.sum() >= 10))
                                    for n, s in systems.items()}

    sc = {n: s["scores"] for n, s in systems.items()}
    if "A800|exact" in sc and "B|exact" in sc:
        prim = compare(sc["A800|exact"], sc["B|exact"], W)
        prim["verdict_hit5"] = verdict(prim["hit5_diff"])
        prim["verdict_mrr"] = verdict(prim["mrr_diff"])
        report["primary"] = {"definition": "B minus A800, exact engine, all queries", **prim}
        ex = report["exploratory"]
        if "B_reviews_only|exact" in sc:
            ex["B_reviews_only_vs_A800"] = compare(sc["A800|exact"], sc["B_reviews_only|exact"], W)
            ex["B_reviews_only_vs_B"] = compare(sc["B|exact"], sc["B_reviews_only|exact"], W)
        if leaky.any():
            ex["B_vs_A800_excluding_leaky"] = compare(sc["A800|exact"], sc["B|exact"], W, ~leaky)
        ex["B_vs_A800_by_type"] = {t: compare(sc["A800|exact"], sc["B|exact"], W, qtype == t)
                                   for t in QUERY_TYPES if (qtype == t).any()}
        ex["depth_sensitivity"] = {}
        for d in sens_depths:
            sa = score_lists(systems["A800|exact"]["rows"], systems["A800|exact"]["v"].outlet_of, gold_sets, depth=d)
            sb = score_lists(systems["B|exact"]["rows"], systems["B|exact"]["v"].outlet_of, gold_sets, depth=d)
            ex["depth_sensitivity"][f"top{d}_chunks"] = compare(sa, sb, W)
    for vn in variants:
        if f"{vn}|chroma" in sc:
            a, b = sc[f"{vn}|exact"], sc[f"{vn}|chroma"]
            report["engine_agreement"][vn] = {
                "queries_where_hit5_differs": int((a["hit5"] != b["hit5"]).sum()),
                "hit5_exact": float(a["hit5"].mean()), "hit5_chroma": float(b["hit5"].mean()),
                "mrr_exact": float(a["rr"].mean()), "mrr_chroma": float(b["rr"].mean())}
    report["caveats"] = [
        "Gold outlets are restricted to outlets with indexed reviews; outlets without reviews exist only in B.",
        "Queries are written by one person: results are about this query set, not about users in general.",
        "Metrics are micro-averaged over queries; check by_type before quoting a single headline number.",
        "McNemar ignores outlet clustering; the cluster bootstrap CI is the primary uncertainty statement.",
        "Hit@k needs k distinct outlets: see short_list_rate_lt5_outlets (retrieval depth is a real cost of clustering).",
    ]

    recs = []
    for n, s in systems.items():
        v, sc_ = s["v"], s["scores"]
        for q in range(len(gold)):
            fg, t1 = int(sc_["first_gold_row"][q]), int(sc_["top1_row"][q])
            recs.append({
                "query_id": gold["query_id"].iat[q], "query_type": qtype[q], "leaky": bool(leaky[q]), "system": n,
                "rank": None if np.isnan(sc_["rank"][q]) else int(sc_["rank"][q]),
                "hit1": int(sc_["hit1"][q]), "hit5": int(sc_["hit5"][q]),
                "rr": float(sc_["rr"][q]), "n_outlets": int(sc_["n_outlets"][q]),
                "outlet_key": gold["gold_outlet_keys"].iat[q][0],
                "first_gold_chunk_type": v.chunk_type[fg] if fg >= 0 else None,
                "first_gold_chunk_id": v.chunk_id[fg] if fg >= 0 else None,
                "top1_chunk_id": v.chunk_id[t1] if t1 >= 0 else None,
                "top1_is_gold": bool(t1 >= 0 and v.outlet_of[t1] in gold_sets[q]),
                "top1_snippet": (v.text[t1][:200] if (v.text is not None and t1 >= 0) else None),
            })
    return report, pd.DataFrame(recs)


def format_summary(report: dict) -> str:
    def cell(d):
        return "n/a" if d["mean"] is None else (f"{d['mean']:.3f} [{d['ci'][0]:.3f}, {d['ci'][1]:.3f}]" if d["ci"]
                                                else f"{d['mean']:.3f}")
    m = report["meta"]
    lines = [f"queries={m['n_queries']} (outlet clusters={m['n_outlet_clusters']}, leaky={m['leaky_queries']}), "
             f"types={m['query_type_counts']}", ""]
    lines.append(f"{'system':<20}{'Hit@1':<26}{'Hit@5':<26}{'MRR':<26}short<5")
    for n, s in report["systems"].items():
        lines.append(f"{n:<20}{cell(s['Hit@1']):<26}{cell(s['Hit@5']):<26}{cell(s['MRR']):<26}"
                     f"{s['short_list_rate_lt5_outlets']:.2f}")
    p = report.get("primary")
    if p:
        d, r = p["hit5_diff"], p["mrr_diff"]
        lines += ["", "PRIMARY (B - A800, exact engine):",
                  f"  dHit@5 = {d['mean']:+.3f} [{d['ci'][0]:+.3f}, {d['ci'][1]:+.3f}]  -> {p['verdict_hit5']}"
                  f"   (McNemar p={p['hit5_discordant']['mcnemar_exact_p']:.4f}, "
                  f"B-only {p['hit5_discordant']['b_only']} / A-only {p['hit5_discordant']['a_only']})",
                  f"  dMRR   = {r['mean']:+.3f} [{r['ci'][0]:+.3f}, {r['ci'][1]:+.3f}]  -> {p['verdict_mrr']}"]
    return "\n".join(lines)


# --------------------------------------------------------------------------------- CLI
def check_manifests(manifests: dict) -> None:
    keys = ["model", "model_commit_hash", "max_seq_length", "passage_prefix", "query_prefix", "dim", "normalize"]
    ref_name, ref = next(iter(manifests.items()))
    for n, m in manifests.items():
        diff = {k: (ref.get(k), m.get(k)) for k in keys if ref.get(k) != m.get(k)}
        if diff:
            raise SystemExit(f"Embeddings of {ref_name} and {n} were not built identically: {diff}")


def parse_args(argv=None) -> argparse.Namespace:
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument("--gold", type=Path, default=Path("eval/gold_queries.jsonl"))
    p.add_argument("--chunks-dir", type=Path, default=Path("data/processed"))
    p.add_argument("--index-input", type=Path, default=Path("data/processed/index_input.parquet"))
    p.add_argument("--emb-dir", type=Path, default=Path("data/embeddings"))
    p.add_argument("--chroma-dir", type=Path, default=Path("data/chroma"))
    p.add_argument("--report-dir", type=Path, default=Path("reports"))
    p.add_argument("--run-dir", type=Path, default=Path("data/eval_runs"),
                   help="per-query CSV incl. dataset text snippets (keep out of git)")
    p.add_argument("--variants", nargs="+", default=["B", "A800"], choices=list(VARIANT_FILES))
    p.add_argument("--k-fetch", type=int, default=150, help="chunks retrieved per query before collapsing to outlets")
    p.add_argument("--sens-depths", type=int, nargs="*", default=[50, 100])
    p.add_argument("--leak-ngram", type=int, default=5)
    p.add_argument("--min-queries", type=int, default=30)
    p.add_argument("--n-boot", type=int, default=10_000)
    p.add_argument("--skip-chroma", action="store_true")
    p.add_argument("--device", default=None)
    p.add_argument("--threads", type=int, default=None)
    return p.parse_args(argv)


def main(argv=None) -> int:
    a = parse_args(argv)
    gold = load_gold(a.gold)
    idx = pd.read_parquet(a.index_input, columns=["review_id", "outlet_key", "text", "mojibake_residual"])
    outlets_with_reviews = set(idx["outlet_key"])
    problems = validate_gold(gold, outlets_with_reviews, idx.set_index("review_id"), a.min_queries)
    if problems:
        print(f"GOLD SET REJECTED ({len(problems)} problems):", file=sys.stderr)
        print("\n".join(f"  - {p}" for p in problems), file=sys.stderr)
        return 2

    variants, manifests = {}, {}
    for vn in a.variants:
        path = a.chunks_dir / VARIANT_FILES[vn]
        manifests[vn] = json.loads((a.emb_dir / f"{vn}.manifest.json").read_text())
        df = pd.read_parquet(path, columns=["chunk_id", "outlet_key", "chunk_type", "text"])
        E = np.load(a.emb_dir / f"{vn}.npy")
        verify_alignment(df, E, manifests[vn], sha256_file(path))
        variants[vn] = VariantData(vn, E, df["outlet_key"].to_numpy(), df["chunk_type"].to_numpy(),
                                   df["chunk_id"].to_numpy(), df["text"].to_numpy())
    check_manifests(manifests)

    gold_keys = {k for ks in gold["gold_outlet_keys"] for k in ks}
    corpus = {k: g["text"].tolist() for k, g in idx[idx["outlet_key"].isin(gold_keys)].groupby("outlet_key")}
    if "B" in variants:
        v = variants["B"]
        prof = (v.chunk_type == "profile") & np.isin(v.outlet_of, list(gold_keys))
        for k, t in zip(v.outlet_of[prof], v.text[prof]):
            corpus.setdefault(k, []).append(t)
    leaky = leaky_flags(gold, corpus, a.leak_ngram)

    ref = next(iter(manifests.values()))
    emb = STEmbedder(ref["model"], ref.get("model_commit_hash"), a.device, None, a.threads)
    Q = emb.encode([ref["query_prefix"] + q for q in gold["query"]])

    chroma_rows = {}
    if not a.skip_chroma:
        import chromadb
        client = chromadb.PersistentClient(path=str(a.chroma_dir))
        for vn, v in variants.items():
            try:
                col = client.get_collection(name=f"zomato_{vn}", embedding_function=None)
            except Exception:  # noqa: BLE001
                print(f"[{vn}] no Chroma collection -> exact engine only", file=sys.stderr)
                continue
            if col.count() != len(v.chunk_id):
                raise SystemExit(f"[{vn}] collection has {col.count()} rows, expected {len(v.chunk_id)}.")
            id_to_row = {cid: i for i, cid in enumerate(v.chunk_id)}
            chroma_rows[vn], _ = chroma_search_fn(col, id_to_row)(Q, a.k_fetch)

    report, per_query = evaluate(variants, Q, gold, leaky, outlets_with_reviews, chroma_rows, a.k_fetch,
                                 a.sens_depths, a.n_boot)
    report["meta"].update(gold_file=str(a.gold), gold_sha256=sha256_file(a.gold),
                          model=ref["model"], model_commit_hash=ref.get("model_commit_hash"),
                          created_utc=datetime.now(timezone.utc).isoformat(timespec="seconds"))
    a.report_dir.mkdir(parents=True, exist_ok=True)
    a.run_dir.mkdir(parents=True, exist_ok=True)
    (a.report_dir / "eval_report.json").write_text(json.dumps(report, indent=2))
    per_query.to_csv(a.run_dir / "eval_per_query.csv", index=False, encoding="utf-8-sig")
    print(format_summary(report))
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""
exact_search.py -- brute-force ground truth for retrieval (shared by parity_test.py and, later, evaluate.py)

Embeddings are L2-normalised float32, so cosine == dot product. ~150k x 384 floats is a ~230 MB matrix:
exact search is a matrix multiply, no index and no approximation error.
"""
from __future__ import annotations

import numpy as np


def exact_topk(E: np.ndarray, Q: np.ndarray, k: int, block: int = 64) -> tuple[np.ndarray, np.ndarray]:
    """Exact top-k rows of E for each query row of Q -> (row_idx[m, k], scores[m, k]), best first.

    Ties inside the top-k are ordered by row index. (A tie exactly at the k-th cut-off is resolved arbitrarily
    but deterministically.)
    """
    n = E.shape[0]
    k = min(k, n)
    idx = np.empty((len(Q), k), dtype=np.int64)
    sc = np.empty((len(Q), k), dtype=np.float32)
    for lo in range(0, len(Q), block):
        S = Q[lo:lo + block] @ E.T
        for r in range(S.shape[0]):
            s = S[r]
            cand = np.arange(n) if k >= n else np.argpartition(-s, k - 1)[:k]
            cand = cand[np.lexsort((cand, -s[cand]))]
            idx[lo + r], sc[lo + r] = cand, s[cand]
    return idx, sc


def collapse_to_outlets(rows, outlet_of, n: int) -> list:
    """Chunk rows in rank order -> first `n` unique outlets in rank order (the unit Hit@5 / MRR are scored on)."""
    seen, out = set(), []
    for r in rows:
        o = outlet_of[r]
        if o not in seen:
            seen.add(o)
            out.append(o)
            if len(out) == n:
                break
    return out


def overlap_at_k(a, b, k: int) -> float:
    """|top-k(a) ∩ top-k(b)| / k."""
    return len(set(a[:k]) & set(b[:k])) / k

#!/usr/bin/env python3

#!/usr/bin/env python3
import pandas as pd
import numpy as np

def main():
    print("Loading clean reviews...")
    df = pd.read_parquet("data/processed/reviews_clean.parquet")
    
    # 1. Strict Filtering (No garbage, no spam, no one-liners)
    mask = (
        (df["n_outlets_same_text"] == 1) &  # Exclude copy-paste spam
        (~df["mojibake_residual"]) &        # Exclude unrecoverable text
        (df["n_chars"] >= 50) &             # Must have some meat to answer a query
        (df["n_chars"] <= 1500)             # Exclude massive reviews for eval simplicity
    )
    clean_pool = df[mask].copy()
    
    # 2. Stratified Sampling (Pick 50 random outlets that have at least 3 valid reviews)
    outlet_counts = clean_pool["outlet_id"].value_counts()
    valid_outlets = outlet_counts[outlet_counts >= 3].index
    
    # Set seed for reproducibility (so Claude can reproduce it tomorrow if he wants)
    np.random.seed(42) 
    sampled_outlets = np.random.choice(valid_outlets, size=50, replace=False)
    
    # 3. Pick up to 4 reviews per sampled outlet
    eval_sample = clean_pool[clean_pool["outlet_id"].isin(sampled_outlets)]
    eval_sample = eval_sample.groupby("outlet_id").head(4).reset_index(drop=True)
    
    # 4. Prepare output for manual query drafting
    out_cols = ["outlet_id", "review_id", "rating", "text"]
    eval_sample = eval_sample[out_cols].sort_values(["outlet_id", "review_id"])
    
    # Save as CSV so you can open it in Excel/Google Sheets to add queries
    out_path = "data/processed/eval_sample_draft.csv"
    eval_sample.to_csv(out_path, index=False)
    
    print(f"Sampling complete! Selected {len(eval_sample)} reviews across {len(sampled_outlets)} outlets.")
    print(f"Saved to {out_path}")
    print("Next Step: Open the CSV, add a 'query' column, and write realistic user questions for each review.")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
load_chroma.py -- Phase 3, step 2

chunks_*.parquet + <variant>.npy + <variant>.manifest.json  ->  Chroma collection zomato_<variant>

    python load_chroma.py --variants B A800
    python load_chroma.py --variants B --rebuild --ef-search 400

Rules
  * Chroma is a SERVING index rebuilt from the .npy + manifest; it is never the source of truth.
  * Alignment is verified before anything is written: row count, ordered chunk_id hash, source-file hash, unit norms.
  * Embeddings are passed explicitly and embedding_function=None, so Chroma's default embedder can never run.
  * Identical HNSW settings for every variant (they are recorded in <variant>.load.json) -- a fairness requirement.
  * A collection is either complete and verified, or the run aborts. Partial / stale collections are never reused:
    use --rebuild. Do not run two processes against the same --chroma-dir at once.
  * ef_search must be >= the n_results you query with (parity_test.py asks for 151).

Requires: pandas, pyarrow, numpy, chromadb
"""
from __future__ import annotations

import argparse
import json
import math
import sys
from dataclasses import asdict, dataclass
from datetime import datetime, timezone
from pathlib import Path

import numpy as np
import pandas as pd

from embed_chunks import VARIANT_FILES, check_embeddings, sha256_file, sha256_ids

META_COLS = ["outlet_key", "chunk_type", "chunk_idx", "review_id", "location", "cuisines", "rating", "cost_for_two"]


@dataclass(frozen=True)
class HnswParams:
    space: str = "cosine"
    ef_construction: int = 200
    ef_search: int = 256
    m: int = 16


# ------------------------------------------------------------------------ alignment
def verify_alignment(df: pd.DataFrame, E: np.ndarray, manifest: dict, source_sha: str | None) -> None:
    """The vectors must belong to exactly this chunk table, row for row."""
    if manifest["rows"] != len(df) or E.shape[0] != len(df):
        raise SystemExit(f"Row count mismatch: manifest {manifest['rows']}, chunk table {len(df)}, vectors {E.shape[0]}.")
    if manifest["ids_sha256"] != sha256_ids(df["chunk_id"].tolist()):
        raise SystemExit("chunk_id order/content differs from the table the embeddings were built from.")
    if source_sha is not None and manifest["source_sha256"] != source_sha:
        raise SystemExit("Chunk file changed after embedding (source hash mismatch). Re-embed.")
    check_embeddings(E, len(df), manifest["dim"])


# ------------------------------------------------------------------------- metadata
def _clean_meta_value(v):
    if v is None or v is pd.NA:
        return None
    if isinstance(v, np.generic):
        v = v.item()  # numpy scalars are rejected by Chroma; plain python only
    if isinstance(v, float) and math.isnan(v):
        return None
    return v


def build_metadatas(df: pd.DataFrame) -> list[dict]:
    """Chroma metadata may not contain None/NaN or numpy types -> drop nulls, convert to python scalars."""
    cols = [c for c in META_COLS if c in df.columns]
    out = []
    for rec in df[cols].to_dict("records"):
        m = {}
        for k, v in rec.items():
            v = _clean_meta_value(v)
            if v is not None:
                m[k] = v
        out.append(m)
    return out


# ------------------------------------------------------------------------ collection
def create_collection(client, name: str, p: HnswParams):
    """Create a collection with explicit HNSW settings. Chroma changed how these are passed between versions,
    so try the newer form first and fall back to the legacy metadata keys. Returns (collection, api_form)."""
    errors = []
    attempts = [
        ("configuration", dict(configuration={"hnsw": {"space": p.space, "ef_construction": p.ef_construction,
                                                       "ef_search": p.ef_search, "max_neighbors": p.m}})),
        ("legacy-metadata", dict(metadata={"hnsw:space": p.space, "hnsw:construction_ef": p.ef_construction,
                                           "hnsw:search_ef": p.ef_search, "hnsw:M": p.m})),
    ]
    for form, kwargs in attempts:
        try:
            return client.create_collection(name=name, embedding_function=None, **kwargs), form
        except Exception as exc:  # noqa: BLE001 -- version differences surface as various exception types
            errors.append(f"{form}: {exc!r}")
            try:
                client.delete_collection(name)  # remove a half-created collection before the next attempt
            except Exception:  # noqa: BLE001
                pass
    raise SystemExit("Could not create the collection with explicit HNSW settings:\n  " + "\n  ".join(errors))


def _get_existing(client, name: str):
    try:
        return client.get_collection(name=name, embedding_function=None)
    except Exception:  # noqa: BLE001 -- "not found" type differs between Chroma versions
        return None


def readback_check(col, df: pd.DataFrame, E: np.ndarray, n_check: int = 200, seed: int = 0) -> None:
    """Fetch random rows back out of Chroma: vectors, documents and outlet_key must match what we put in."""
    rng = np.random.default_rng(seed)
    sel = rng.choice(len(df), size=min(n_check, len(df)), replace=False)
    ids = [df["chunk_id"].iat[int(i)] for i in sel]
    res = col.get(ids=ids, include=["embeddings", "documents", "metadatas"])
    pos = {cid: j for j, cid in enumerate(res["ids"])}
    if len(pos) != len(ids):
        raise SystemExit(f"Read-back returned {len(pos)} of {len(ids)} requested ids.")
    emb = np.asarray(res["embeddings"], dtype=np.float32)
    for i, cid in zip(sel, ids):
        j, i = pos[cid], int(i)
        if not np.allclose(emb[j], E[i], atol=1e-6):
            raise SystemExit(f"Read-back: stored vector differs from the .npy for {cid}.")
        if res["documents"][j] != df["text"].iat[i]:
            raise SystemExit(f"Read-back: stored document differs for {cid}.")
        if res["metadatas"][j].get("outlet_key") != df["outlet_key"].iat[i]:
            raise SystemExit(f"Read-back: stored outlet_key differs for {cid}.")


def load_variant(client, variant: str, df: pd.DataFrame, E: np.ndarray, params: HnswParams,
                 rebuild: bool = False, batch: int = 5000, log=print) -> dict:
    name, n = f"zomato_{variant}", len(df)
    col = _get_existing(client, name)
    if col is not None and rebuild:
        client.delete_collection(name)
        col = None
    if col is not None:
        have = col.count()
        if have != n:
            raise SystemExit(f"[{variant}] collection {name} holds {have} of {n} rows (partial or stale). "
                             f"Re-run with --rebuild.")
        readback_check(col, df, E)
        log(f"[{variant}] {name} already loaded ({n:,} rows) and verified -> skipping")
        return {"collection": name, "rows": n, "skipped": True}

    col, api_form = create_collection(client, name, params)
    ids, docs, metas = df["chunk_id"].tolist(), df["text"].tolist(), build_metadatas(df)
    max_bs = int(getattr(client, "get_max_batch_size", lambda: 5000)())
    bs = max(1, min(batch, max_bs))
    for lo in range(0, n, bs):
        hi = min(n, lo + bs)
        col.upsert(ids=ids[lo:hi], embeddings=E[lo:hi].tolist(), documents=docs[lo:hi], metadatas=metas[lo:hi])
        log(f"[{variant}] upserted {hi:,}/{n:,}")
    if col.count() != n:
        raise SystemExit(f"[{variant}] count after load {col.count()} != {n}.")
    readback_check(col, df, E)
    log(f"[{variant}] {name} loaded and verified ({n:,} rows, HNSW via {api_form})")
    return {"collection": name, "rows": n, "skipped": False, "api_form": api_form, "batch_size": bs}


# ----------------------------------------------------------------------------------- CLI
def parse_args(argv=None) -> argparse.Namespace:
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument("--chunks-dir", type=Path, default=Path("data/processed"))
    p.add_argument("--emb-dir", type=Path, default=Path("data/embeddings"))
    p.add_argument("--chroma-dir", type=Path, default=Path("data/chroma"))
    p.add_argument("--variants", nargs="+", default=["B", "A800"], choices=[*VARIANT_FILES, "all"])
    p.add_argument("--rebuild", action="store_true", help="drop and recreate the collection(s)")
    p.add_argument("--space", default="cosine", choices=["cosine", "l2", "ip"])
    p.add_argument("--ef-construction", type=int, default=200)
    p.add_argument("--ef-search", type=int, default=256)
    p.add_argument("--m", type=int, default=16)
    p.add_argument("--batch", type=int, default=5000)
    return p.parse_args(argv)


def main(argv=None) -> int:
    import chromadb

    a = parse_args(argv)
    params = HnswParams(a.space, a.ef_construction, a.ef_search, a.m)
    variants = list(VARIANT_FILES) if "all" in a.variants else a.variants
    a.chroma_dir.mkdir(parents=True, exist_ok=True)
    client = chromadb.PersistentClient(path=str(a.chroma_dir))

    for v in variants:
        chunk_path = a.chunks_dir / VARIANT_FILES[v]
        npy, man = a.emb_dir / f"{v}.npy", a.emb_dir / f"{v}.manifest.json"
        for pth in (chunk_path, npy, man):
            if not pth.exists():
                raise SystemExit(f"{pth} not found (embeddings for {v} not finished?)")
        manifest = json.loads(man.read_text())
        df = pd.read_parquet(chunk_path)
        E = np.load(npy)
        verify_alignment(df, E, manifest, sha256_file(chunk_path))
        info = load_variant(client, v, df, E, params, a.rebuild, a.batch)
        info.update(variant=v, hnsw=asdict(params), chroma_version=getattr(chromadb, "__version__", None),
                    embeddings_ids_sha256=manifest["ids_sha256"],
                    model_commit_hash=manifest.get("model_commit_hash"),
                    loaded_utc=datetime.now(timezone.utc).isoformat(timespec="seconds"))
        if not info.get("skipped"):
            (a.emb_dir / f"{v}.load.json").write_text(json.dumps(info, indent=2))
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""Convert the annotated draft CSV into eval/gold_queries.jsonl.

Input: data/processed/eval_sample_draft.csv with `query` and `query_type` filled in
(see fill_queries.py). Each row's gold outlet is looked up by review_id in
data/processed/index_input.parquet, so gold_outlet_keys holds the 12-hex outlet_key
that evaluate.py expects. The CSV's own outlet_id column (the long name+address
string) is not used.

Output schema (one JSON object per line):
  query_id, query, query_type, gold_outlet_keys (list of one 12-hex key), source_review_id

Rows with a blank `query` are skipped. Nothing is written if any check fails.
"""
import json
import os
import re
import sys

import pandas as pd

DRAFT = "data/processed/eval_sample_draft.csv"
INDEX = "data/processed/index_input.parquet"
OUT = "eval/gold_queries.jsonl"
MIN_QUERIES = 30  # same threshold as validate_gold
KEY_RE = re.compile(r"[0-9a-fA-F]{12}")


def main():
    # Everything is read as text so IDs can't be turned into floats or lose characters.
    df = pd.read_csv(DRAFT, dtype=str, keep_default_na=False, encoding="utf-8-sig")
    for col in ("review_id", "query", "query_type"):
        if col not in df.columns:
            sys.exit(f"Draft CSV has no '{col}' column (columns: {list(df.columns)}).")

    df["query"] = df["query"].str.strip()
    df["query_type"] = df["query_type"].str.strip()
    gold = df[df["query"] != ""].copy().reset_index(drop=True)

    # review_id -> outlet_key, from the same table the index is built from.
    # drop_duplicates() so a chunk-level table (several rows per review) is fine.
    idx = pd.read_parquet(INDEX, columns=["review_id", "outlet_key"]).astype(str).drop_duplicates()
    ambiguous = idx.loc[idx["review_id"].duplicated(keep=False), "review_id"].unique()
    if len(ambiguous):
        sys.exit(f"{len(ambiguous)} review_ids map to more than one outlet_key in {INDEX}, "
                 f"e.g. {list(ambiguous[:3])}. Nothing written.")
    key_of = dict(zip(idx["review_id"], idx["outlet_key"]))
    gold["outlet_key"] = gold["review_id"].map(key_of)

    # Local pre-flight checks that mirror validate_gold.
    problems = []
    if len(gold) < MIN_QUERIES:
        problems.append(f"only {len(gold)} queries (< {MIN_QUERIES})")

    no_type = gold[gold["query_type"] == ""]
    if len(no_type):
        problems.append(f"{len(no_type)} rows have a query but no query_type "
                        f"(first: review_id {no_type['review_id'].iloc[0]})")

    norm = gold["query"].str.lower().str.replace(r"\W+", " ", regex=True).str.strip()
    for q in gold.loc[norm.duplicated(keep=False), "query"].unique():
        problems.append(f"duplicate query text: {q!r}")

    not_indexed = gold[gold["outlet_key"].isna()]
    if len(not_indexed):
        problems.append(f"{len(not_indexed)} review_ids not found in {INDEX}, e.g. "
                        f"{not_indexed['review_id'].head(3).tolist()} "
                        "(spreadsheet apps can mangle IDs; re-export the CSV as text)")

    bad_key = gold[gold["outlet_key"].notna() & ~gold["outlet_key"].fillna("").str.fullmatch(KEY_RE)]
    if len(bad_key):
        problems.append(f"{len(bad_key)} outlet_keys are not 12-hex, e.g. "
                        f"{bad_key['outlet_key'].head(2).tolist()}")

    if problems:
        print("Not writing gold file. Fix these first:")
        for p in problems:
            print(f"  - {p}")
        sys.exit(1)

    os.makedirs(os.path.dirname(OUT), exist_ok=True)
    with open(OUT, "w", encoding="utf-8") as f:
        rows = zip(gold["outlet_key"], gold["review_id"], gold["query"], gold["query_type"])
        for i, (key, review_id, query, qtype) in enumerate(rows, 1):
            f.write(json.dumps({
                "query_id": f"q{i:03d}",
                "query": query,
                "query_type": qtype,
                "gold_outlet_keys": [key],
                "source_review_id": review_id,
            }, ensure_ascii=False) + "\n")

    print(f"Wrote {len(gold)} queries across {gold['outlet_key'].nunique()} outlets to {OUT}")
    print(gold["query_type"].value_counts().to_string())


if __name__ == "__main__":
    main()

"""
outlet_key.py - the ONE definition of outlet identity.
"""
from __future__ import annotations
from typing import Sequence
import pandas as pd

SEP = "\x1f"

def _clean(s: pd.Series) -> pd.Series:
    out = (
        s.astype("string")
        .str.normalize("NFKC")
        .str.replace(r"\s+", " ", regex=True)
        .str.strip()
        .str.casefold()
    )
    return out.mask(out == "")

def build_outlet_id(df: pd.DataFrame, cols: Sequence[str]) -> pd.Series:
    cols = list(cols)
    if not cols:
        raise ValueError("need at least one key column")
    missing = [c for c in cols if c not in df.columns]
    if missing:
        raise KeyError(f"key column(s) not in data: {missing}; have: {list(df.columns)}")
    out = _clean(df[cols[0]])
    for c in cols[1:]:
        out = out + SEP + _clean(df[c])
    return out

#!/usr/bin/env python3
"""
parity_test.py -- Phase 3, step 2b: does Chroma's approximate (HNSW) search return what exact search returns?

    python parity_test.py --variants B A800

Queries are random chunk vectors from the variant itself (no model needed). For each query both engines fetch the
top-K; then, identically on both sides:
  * chunk level : the query's own chunk is dropped; recall@10 / recall@50 of Chroma vs exact
  * outlet level: the query's own OUTLET is dropped, the rest is collapsed to unique outlets in rank order,
                  and the top-5 outlets are compared. This is the unit Hit@5 / MRR are scored on, so it is the
                  number that decides whether ANN error can contaminate the A-vs-B comparison.
Exit code 3 if a gate fails (report is still written to --report-dir). Same gates for every variant.
"""
from __future__ import annotations

import argparse
import json
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

import numpy as np
import pandas as pd

from embed_chunks import VARIANT_FILES, sha256_file
from exact_search import collapse_to_outlets, exact_topk, overlap_at_k
from load_chroma import verify_alignment


def infer_metric(E, qi, ch_rows, ch_dist, n_queries: int = 20, tol: float = 1e-3) -> str:
    """Chroma reports 1-dot for cosine/ip and squared L2 (2-2*dot) for l2, on unit vectors. Ranking is identical
    for all three on normalised embeddings; this only tells you which one the collection actually uses."""
    dots, ds = [], []
    for q in range(min(n_queries, len(qi))):
        for r, d in zip(ch_rows[q][:5], ch_dist[q][:5]):
            dots.append(float(E[qi[q]] @ E[r]))
            ds.append(float(d))
    dots, ds = np.array(dots), np.array(ds)
    if np.allclose(ds, 1 - dots, atol=tol):
        return "cosine_or_ip"
    if np.allclose(ds, 2 - 2 * dots, atol=tol):
        return "l2_squared"
    return "unknown"


def run_parity(search_fn, E: np.ndarray, outlet_of, n_queries: int = 500, k_fetch: int = 150, seed: int = 0,
               n_outlets: int = 5) -> dict:
    """search_fn(Q, k) -> (rows[list[list[int]]], distances[list[list[float]]]) for the approximate engine."""
    rng = np.random.default_rng(seed)
    qi = np.sort(rng.choice(len(E), size=min(n_queries, len(E)), replace=False))
    Q = E[qi]

    t = time.perf_counter()
    ex_idx, _ = exact_topk(E, Q, k_fetch + 1)
    exact_ms = (time.perf_counter() - t) * 1000 / len(qi)
    t = time.perf_counter()
    ch_rows, ch_dist = search_fn(Q, k_fetch + 1)
    ann_ms = (time.perf_counter() - t) * 1000 / len(qi)

    r10, r50, top1, o_ov, o_top1, short = [], [], [], [], [], 0
    for q, row in enumerate(qi):
        own = outlet_of[row]
        ex = [int(r) for r in ex_idx[q] if r != row]
        ch = [int(r) for r in ch_rows[q] if r != row]
        r10.append(overlap_at_k(ex, ch, 10))
        r50.append(overlap_at_k(ex, ch, 50))
        top1.append(float(ex[:1] == ch[:1]))
        ex_o = collapse_to_outlets([r for r in ex if outlet_of[r] != own], outlet_of, n_outlets)
        ch_o = collapse_to_outlets([r for r in ch if outlet_of[r] != own], outlet_of, n_outlets)
        short += len(ch_o) < n_outlets
        o_ov.append(len(set(ex_o) & set(ch_o)) / n_outlets)
        o_top1.append(float(ex_o[:1] == ch_o[:1]))

    return {
        "n_queries": int(len(qi)), "k_fetch": k_fetch,
        "chunk_recall@10_mean": float(np.mean(r10)), "chunk_recall@10_min": float(np.min(r10)),
        "chunk_recall@50_mean": float(np.mean(r50)), "chunk_top1_agreement": float(np.mean(top1)),
        f"outlet_top{n_outlets}_overlap_mean": float(np.mean(o_ov)),
        f"outlet_top{n_outlets}_overlap_min": float(np.min(o_ov)),
        "outlet_top1_agreement": float(np.mean(o_top1)),
        "queries_with_short_ann_outlet_list": int(short),
        "metric_inferred": infer_metric(E, qi, ch_rows, ch_dist),
        "exact_ms_per_query": round(exact_ms, 2), "ann_ms_per_query": round(ann_ms, 2),
    }


def evaluate_gates(res: dict, min_recall10: float, min_outlet_overlap: float, n_outlets: int = 5) -> dict:
    checks = {
        f"chunk_recall@10_mean >= {min_recall10}": res["chunk_recall@10_mean"] >= min_recall10,
        f"outlet_top{n_outlets}_overlap_mean >= {min_outlet_overlap}":
            res[f"outlet_top{n_outlets}_overlap_mean"] >= min_outlet_overlap,
        "no short ANN outlet lists": res["queries_with_short_ann_outlet_list"] == 0,
    }
    return {"checks": checks, "passed": all(checks.values())}


def chroma_search_fn(col, id_to_row: dict, block: int = 100):
    def fn(Q, k):
        rows, dists = [], []
        for lo in range(0, len(Q), block):
            res = col.query(query_embeddings=Q[lo:lo + block].tolist(), n_results=k, include=["distances"])
            for ids, ds in zip(res["ids"], res["distances"]):
                rows.append([id_to_row[i] for i in ids])
                dists.append(ds)
        return rows, dists
    return fn


def parse_args(argv=None) -> argparse.Namespace:
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument("--chunks-dir", type=Path, default=Path("data/processed"))
    p.add_argument("--emb-dir", type=Path, default=Path("data/embeddings"))
    p.add_argument("--chroma-dir", type=Path, default=Path("data/chroma"))
    p.add_argument("--report-dir", type=Path, default=Path("reports"))
    p.add_argument("--variants", nargs="+", default=["B", "A800"], choices=[*VARIANT_FILES, "all"])
    p.add_argument("--n-queries", type=int, default=500)
    p.add_argument("--k-fetch", type=int, default=150, help="results fetched per query (needs ef_search > this)")
    p.add_argument("--min-recall10", type=float, default=0.98)
    p.add_argument("--min-outlet-overlap", type=float, default=0.98)
    return p.parse_args(argv)


def main(argv=None) -> int:
    import chromadb

    a = parse_args(argv)
    variants = list(VARIANT_FILES) if "all" in a.variants else a.variants
    a.report_dir.mkdir(parents=True, exist_ok=True)
    client = chromadb.PersistentClient(path=str(a.chroma_dir))
    all_ok = True

    for v in variants:
        chunk_path = a.chunks_dir / VARIANT_FILES[v]
        manifest = json.loads((a.emb_dir / f"{v}.manifest.json").read_text())
        df = pd.read_parquet(chunk_path, columns=["chunk_id", "outlet_key"])
        E = np.load(a.emb_dir / f"{v}.npy")
        verify_alignment(df, E, manifest, sha256_file(chunk_path))

        col = client.get_collection(name=f"zomato_{v}", embedding_function=None)
        if col.count() != len(df):
            raise SystemExit(f"[{v}] collection holds {col.count()} rows, expected {len(df)}. Run load_chroma.py --rebuild.")
        load_info = {}
        lp = a.emb_dir / f"{v}.load.json"
        if lp.exists():
            load_info = json.loads(lp.read_text())
            ef = load_info.get("hnsw", {}).get("ef_search", 0)
            if ef <= a.k_fetch:
                print(f"[{v}] WARNING: ef_search={ef} <= k_fetch={a.k_fetch}; Chroma may error or truncate results.",
                      file=sys.stderr)

        id_to_row = {cid: i for i, cid in enumerate(df["chunk_id"])}
        res = run_parity(chroma_search_fn(col, id_to_row), E, df["outlet_key"].to_numpy(), a.n_queries, a.k_fetch)
        gates = evaluate_gates(res, a.min_recall10, a.min_outlet_overlap)
        report = {"variant": v, "rows": len(df), "hnsw": load_info.get("hnsw"),
                  "chroma_version": load_info.get("chroma_version"), **res, **gates,
                  "created_utc": datetime.now(timezone.utc).isoformat(timespec="seconds")}
        (a.report_dir / f"parity_{v}.json").write_text(json.dumps(report, indent=2))
        print(json.dumps(report, indent=2))
        all_ok &= gates["passed"]

    if not all_ok:
        print("PARITY GATE FAILED: raise --ef-search / --ef-construction in load_chroma.py (--rebuild), identically "
              "for every variant, and re-run.", file=sys.stderr)
        return 3
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
from __future__ import annotations

import argparse
import ast
import hashlib
import json
import re
import sys
from functools import lru_cache
from pathlib import Path

import ftfy
import pandas as pd
from ftfy.badness import is_bad

# کلاؤڈ کی الگ کی گئی فائل سے امپورٹ
from outlet_key import build_outlet_id

RATED_PREFIX = re.compile(r"^\s*RATED\b\s*")
CTRL_CHARS = re.compile(r"[\x00-\x08\x0b-\x1f\x7f]")
WS = re.compile(r"\s+")
RATING_NUM = re.compile(r"\d+(?:\.\d+)?")
OVERSIZED_CHARS = 20_000

def parse_reviews_list(raw: str) -> tuple[list[tuple[str | None, str]], str | None]:
    raw = raw.strip()
    if raw in ("", "[]"): return [], None
    try: obj = ast.literal_eval(raw)
    except (ValueError, SyntaxError, MemoryError, RecursionError) as exc: return [], f"literal_eval:{type(exc).__name__}"
    if not isinstance(obj, (list, tuple)): return [], "not_a_list"
    items, bad = [], 0
    for it in obj:
        if isinstance(it, (list, tuple)) and len(it) == 2 and isinstance(it[1], str):
            items.append((it[0] if isinstance(it[0], str) else None, it[1]))
        else: bad += 1
    return items, (f"bad_items:{bad}" if bad else None)

def parse_rating(s: str | None) -> float:
    if not isinstance(s, str): return float("nan")
    m = RATING_NUM.search(s)
    return float(m.group()) if m else float("nan")

@lru_cache(maxsize=None)
def clean_text(raw: str) -> tuple[str, bool, bool]:
    if raw.isascii(): t, before, after = raw, False, False
    else:
        before = is_bad(raw) or "\ufffd" in raw
        t = ftfy.fix_text(raw)
        after = is_bad(t) or "\ufffd" in t
        t = t.encode("utf-8", "replace").decode("utf-8")
    t = RATED_PREFIX.sub("", t, count=1)
    t = CTRL_CHARS.sub(" ", t)
    t = WS.sub(" ", t).strip()
    return t, before, after

def explode_and_dedup(df: pd.DataFrame, key: str) -> tuple[pd.DataFrame, list[dict], dict]:
    funnel: dict = {"raw_rows": len(df), "raw_rows_null_reviews": int(df["reviews_list"].isna().sum())}
    df = df.assign(reviews_list=df["reviews_list"].fillna("[]"))
    funnel["outlets_in_raw"] = int(df[key].nunique())
    pairs = df.groupby([key, "reviews_list"], sort=False).size().rename("n_rows").reset_index()
    funnel["distinct_outlet_reviewlist_pairs"] = len(pairs)
    rejects: list[dict] = []
    recs: list[tuple] = []
    for pair_id, (oid, raw, n_rows) in enumerate(pairs.itertuples(index=False, name=None)):
        items, err = parse_reviews_list(raw)
        if err: rejects.append({"outlet_id": oid, "error": err, "raw_len": len(raw), "raw_head": raw[:300]})
        recs.extend((oid, pair_id, n_rows, pos, r, t) for pos, (r, t) in enumerate(items))
    funnel["parse_error_pairs"] = len(rejects)
    if not recs: raise SystemExit("No reviews parsed at all -- wrong column or wrong file?")
    ex = pd.DataFrame(recs, columns=["outlet_id", "pair_id", "n_rows", "pos", "rating_raw", "text_raw"])
    ex["seq"] = range(len(ex))
    funnel["reviews_exploded"] = len(ex)
    cleaned = [clean_text(t) for t in ex["text_raw"]]
    ex["text"] = [c[0] for c in cleaned]
    ex["moji_before"] = [c[1] for c in cleaned]
    ex["moji_after"] = [c[2] for c in cleaned]
    ex["rating"] = [parse_rating(r) for r in ex["rating_raw"]]
    ex = ex.drop(columns=["rating_raw", "text_raw"])
    empty = ex["text"].str.len() == 0
    funnel["dropped_empty_text"] = int(empty.sum())
    ex = ex[~empty].copy()
    ex["text_key"] = ex["text"].str.casefold()
    ex = ex.drop_duplicates(["outlet_id", "text_key", "pair_id"])
    out = (ex.groupby(["outlet_id", "text_key"], sort=False).agg(
        text=("text", "first"), rating=("rating", "first"), rating_nuniq=("rating", "nunique"),
        source_rows=("n_rows", "sum"), mojibake_before=("moji_before", "max"),
        mojibake_residual=("moji_after", "max"), seq=("seq", "min")
    ).reset_index())
    out["rating_conflict"] = out.pop("rating_nuniq") > 1
    out["review_order"] = out.groupby("outlet_id")["seq"].rank(method="first").astype("int64") - 1
    out["n_chars"] = out["text"].str.len().astype("int64")
    out["n_words"] = out["text"].str.split().str.len().astype("int64")
    out["is_oversized"] = out["n_chars"] > OVERSIZED_CHARS
    out["n_outlets_same_text"] = out.groupby("text_key")["outlet_id"].transform("nunique").astype("int64")
    out["review_id"] = [hashlib.sha1(f"{o}\x1f{k}".encode("utf-8")).hexdigest()[:16] for o, k in zip(out["outlet_id"], out["text_key"])]
    cols = ["review_id", "outlet_id", "review_order", "rating", "text", "n_chars", "n_words",
            "source_rows", "n_outlets_same_text", "rating_conflict", "mojibake_before", "mojibake_residual", "is_oversized"]
    out = out.sort_values("seq", kind="stable")[cols].reset_index(drop=True)
    q = out["n_chars"].quantile([0.5, 0.9, 0.99, 0.999])
    funnel.update(
        reviews_after_dedup=len(out), duplicates_removed=funnel["reviews_exploded"] - funnel["dropped_empty_text"] - len(out),
        outlets_with_reviews=int(out["outlet_id"].nunique()), mojibake_before_share=round(float(out["mojibake_before"].mean()), 4),
        mojibake_residual_share=round(float(out["mojibake_residual"].mean()), 4), rating_conflicts=int(out["rating_conflict"].sum()),
        oversized=int(out["is_oversized"].sum()), cross_outlet_dup_reviews=int((out["n_outlets_same_text"] > 1).sum()),
        n_chars_quantiles={f"p{round(k * 1000) / 10:g}": int(v) for k, v in q.items()}, n_chars_max=int(out["n_chars"].max()),
    )
    return out, rejects, funnel

def validate(df: pd.DataFrame, failures_path: Path) -> pd.DataFrame:
    try: import pandera.pandas as pa
    except ImportError: import pandera as pa
    schema = pa.DataFrameSchema({
        "review_id": pa.Column(str, pa.Check.str_matches(r"^[0-9a-f]{16}$"), unique=True),
        "outlet_id": pa.Column(str, pa.Check.str_length(min_value=1)), "review_order": pa.Column(int, pa.Check.ge(0)),
        "rating": pa.Column(float, pa.Check.in_range(0.0, 5.0), nullable=True),
        "text": pa.Column(str, checks=[pa.Check(lambda s: ~s.str.match(r"^\s*RATED\b"), error="RATED prefix leaked"), pa.Check(lambda s: ~s.str.contains(CTRL_CHARS.pattern), error="control chars left"), pa.Check(lambda s: s == s.str.strip(), error="untrimmed text"), pa.Check(lambda s: ~s.str.contains(r"\s{2,}"), error="whitespace not collapsed"), pa.Check(lambda s: s.str.len() > 0, error="empty text")]),
        "n_chars": pa.Column(int, pa.Check.ge(1)), "n_words": pa.Column(int, pa.Check.ge(1)),
        "source_rows": pa.Column(int, pa.Check.ge(1)), "n_outlets_same_text": pa.Column(int, pa.Check.ge(1)),
        "rating_conflict": pa.Column(bool), "mojibake_before": pa.Column(bool), "mojibake_residual": pa.Column(bool), "is_oversized": pa.Column(bool),
    }, checks=[pa.Check(lambda d: d["n_words"] <= d["n_chars"], error="n_words > n_chars"), pa.Check(lambda d: d["n_chars"] == d["text"].str.len(), error="n_chars stale")], unique=["outlet_id", "review_order"], strict=True, coerce=True)
    try: return schema.validate(df, lazy=True)
    except pa.errors.SchemaErrors as exc:
        exc.failure_cases.to_csv(failures_path, index=False)
        raise SystemExit(f"Pandera validation FAILED: {len(exc.failure_cases)} failure cases -> {failures_path}\n{exc.failure_cases.groupby(['column', 'check']).size().to_string()}")

def check_outlets(reviews: pd.DataFrame, path: Path, keys: list[str]) -> dict:
    ref_df = pd.read_parquet(path, columns=keys)
    ref_ids = set(build_outlet_id(ref_df, keys).dropna())
    have = set(reviews["outlet_id"])
    n_blank = reviews["outlet_id"].isna().sum()
    if n_blank: raise SystemExit(f"{n_blank} reviews have a blank key column; refusing to guess")
    orphans = have - ref_ids
    if orphans: raise SystemExit(f"{len(orphans)} outlet_ids in reviews are not in Day-1 outlets (e.g. {sorted(orphans)[:3]!r})")
    return {"day1_outlets": len(ref_ids), "day1_outlets_without_reviews": len(ref_ids - have)}

def parse_args(argv=None) -> argparse.Namespace:
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument("--raw", required=True, type=Path)
    p.add_argument("--key", default="name,address", type=lambda s: [c.strip() for c in s.split(",") if c.strip()])
    p.add_argument("--outlets", type=Path)
    p.add_argument("--encoding", default="utf-8")
    p.add_argument("--out-dir", type=Path, default=Path("data/processed"))
    p.add_argument("--max-parse-error-rate", type=float, default=0.001)
    return p.parse_args(argv)

def main(argv=None) -> int:
    a = parse_args(argv)
    a.out_dir.mkdir(parents=True, exist_ok=True)
    use_cols = a.key + ["reviews_list"]
    raw = pd.read_csv(a.raw, usecols=use_cols, dtype=str, encoding=a.encoding)
    
    raw["outlet_id"] = build_outlet_id(raw, a.key)
    n_null_key = int(raw["outlet_id"].isna().sum())
    raw = raw.dropna(subset=["outlet_id"])

    reviews, rejects, funnel = explode_and_dedup(raw, "outlet_id")
    funnel["raw_rows_null_key_dropped"] = n_null_key
    if rejects: pd.DataFrame(rejects).to_parquet(a.out_dir / "reviews_rejects.parquet", index=False)
    err_rate = funnel["parse_error_pairs"] / funnel["distinct_outlet_reviewlist_pairs"]
    if err_rate > a.max_parse_error_rate: return 2

    reviews = validate(reviews, a.out_dir / "reviews_validation_failures.csv")
    if a.outlets: funnel.update(check_outlets(reviews, a.outlets, a.key))

    reviews.to_parquet(a.out_dir / "reviews_clean.parquet", index=False, compression="zstd")
    reviews.to_json(a.out_dir / "reviews_clean.jsonl", orient="records", lines=True, force_ascii=False)
    (a.out_dir / "reviews_prep_report.json").write_text(json.dumps(funnel, indent=2))
    print(json.dumps(funnel, indent=2))
    return 0

if __name__ == "__main__":
    sys.exit(main())

# Zomato Bangalore: Data Prep Pipeline

Collapses the Kaggle Zomato Bangalore Restaurants dataset to one record per outlet, cleans `rate` and `approx_cost`, validates the result with Pandera, and exports Parquet and JSONL for analytics or downstream LLM ingestion.

## Finding
Exact-row `drop_duplicates()` removes 0 rows because every row has a unique `url` (51,717 unique values in 51,717 rows). The same outlet repeats across many rows (different `listed_in(type)` values, vote counts and URLs), so the 51,717 rows describe 12,499 outlets when keyed on `name` + `address`.

After normalizing formatting (`3.9/5` vs `3.9 /5`, newlines in phone numbers), rows of the same outlet still disagree on `rate`, `votes` or `approx_cost` for 4,277 of the 12,499 outlets. The cause was not verified.

## Pipeline
1. Standardize column names
2. Collapse to one row per outlet (key: `name` + `address`): keep the highest-vote row, set `location` to the most frequent label, keep `listed_intype` / `listed_incity` as lists
3. Add `is_new` flag; convert `NEW` / `-` ratings to null
4. Parse `rate` and `approx_cost` to numeric
5. Validate with Pandera
6. Export Parquet and JSONL

## Schema rules
`rate` between 0 and 5, `approx_cost` > 0, non-null `name` and `address`.

## Limitations
- `name` + `address` is a heuristic key; the same outlet with a differently spelled address would not merge.
- Only the highest-vote row is kept; its `rate` (and so `is_new`) is used even if a sibling row has a different or non-missing rating.
- `location` differs across rows for 28 outlets; the most frequent label is kept.
- `phone` differs across rows for some outlets and is not normalized.
- Only `rate`, `approx_cost` and `votes` are cleaned; `phone`, `cuisines`, `dish_liked`, `reviews_list`, `menu_item` and other columns are carried over as-is.
- JSONL is not chunked or embedded; it is input to a RAG pipeline, not a finished one.
- The CSV and the generated Parquet/JSONL files are not committed.

Data: Kaggle, Zomato Bangalore Restaurants.

## Conclusion

**Strategy B (structured chunking with profile metadata) ranks the correct outlet higher than fixed 800-word chunks (A800), and the ranking gain is statistically clear. The top-5 recall gain points the same way but is not conclusive at this sample size.**

### Results

50 gold queries, each with one gold outlet. 95% confidence intervals in brackets.

| System | Hit@1 | Hit@5 | MRR |
|---|---|---|---|
| B (exact) | 0.240 [0.154, 0.327] | 0.400 [0.294, 0.510] | 0.322 [0.237, 0.405] |
| B (Chroma) | 0.220 [0.122, 0.314] | 0.380 [0.255, 0.500] | 0.303 [0.206, 0.396] |
| B_restricted (exact) | 0.240 [0.154, 0.327] | 0.400 [0.294, 0.510] | 0.323 [0.238, 0.406] |
| A800 (exact) | 0.100 [0.020, 0.196] | 0.260 [0.104, 0.431] | 0.183 [0.082, 0.294] |
| A800 (Chroma) | 0.100 [0.020, 0.196] | 0.260 [0.128, 0.400] | 0.179 [0.084, 0.283] |

Primary comparison (B - A800, exact engine):

- **MRR: +0.139 [+0.072, +0.211].** The interval excludes 0, so B ranks the right outlet higher.
- **Hit@5: +0.140 [-0.019, +0.275].** Inconclusive, because the interval includes 0 (McNemar p = 0.065). B alone found the gold outlet in the top 5 on 9 queries, A800 alone on 2.

### Takeaways

- **Ranking quality:** B places the correct outlet higher on average than A800 (MRR 0.32 vs 0.18).
- **Vector store:** Chroma's approximate index tracks brute-force exact search closely (differences of 0.02 or less on Hit@1, Hit@5 and MRR for both strategies), so the comparison between strategies is not an artefact of the vector store.

### Limitations

- **Small, narrow gold set:** 50 queries drawn from 13 outlets. Each query has a single gold outlet, so a query that other outlets could also answer counts as a miss, which makes the recall figures conservative.
- **LLM-drafted queries:** The queries were drafted with an LLM from sampled reviews and may not match how real users phrase questions.
- **Modest absolute recall:** Even B surfaces the gold outlet in the top 5 for only 40% of queries.
- **Bundled changes:** B combines structured chunking with profile metadata, so these results do not separate the contribution of each.

### Next steps

- Grow the gold set to about 150 queries spread across all sampled outlets. That would narrow the Hit@5 interval by roughly 40%.
- Run chunking-only and metadata-only variants to attribute the gain.
- Add a keyword (BM25) hybrid or a reranker and re-run the same harness.

"""pytest -q test_build_chunks.py   (no model download: uses a whitespace 'tokenizer')"""
import random
import re
import unicodedata

import numpy as np

class FakeCounter:
    """1 token == 1 whitespace-separated word. Same interface as HFCounter."""

    def count(self, texts, batch=0):
        return np.array([len(t.split()) for t in texts], dtype="int64")

    def offsets(self, text):
        return [m.span() for m in re.finditer(r"\S+", text)]


C = FakeCounter()


class BertLikeCounter:
    """Emulates how a BERT-uncased fast tokenizer treats odd characters: control/format chars (Cc/Cf, e.g. ZWSP, ZWJ),
    U+FFFD and combining marks (Mn, e.g. emoji variation selector U+FE0F) are DROPPED -- no token, no offset.
    Punctuation is its own token; whitespace separates words."""

    def _toks(self, text):
        toks, cur = [], None
        for i, ch in enumerate(text):
            cat = unicodedata.category(ch)
            if ch == "\ufffd" or (cat.startswith("C") and ch not in "\t\n\r") or cat == "Mn":
                continue
            if ch.isspace():
                cur = None
                continue
            if cat.startswith("P") or (ord(ch) < 128 and not ch.isalnum()):
                toks.append([i, i + 1])
                cur = None
                continue
            if cur is None:
                cur = [i, i + 1]
                toks.append(cur)
            else:
                cur[1] = i + 1
        return [tuple(t) for t in toks]

    def count(self, texts, batch=0):
        return np.array([len(self._toks(t)) for t in texts], dtype="int64")

    def offsets(self, text):
        return self._toks(text)


BC = BertLikeCounter()

"""pytest -q test_embed_chunks.py   (no model download: deterministic fake embedder)"""
import hashlib
import json
import tempfile
from pathlib import Path

import numpy as np
import pandas as pd

import embed_chunks as ec

DIM = 16


class FakeEmbedder:
    """Deterministic per-text unit vectors. Counts encode() calls/rows so tests can prove what was re-computed."""

    def __init__(self, commit="abc123", fail_after_calls=None, nan=False, nondeterministic=False, max_seq=512):
        self.device, self.dim, self.max_seq_length = "cpu", DIM, max_seq
        self.info = {"model": "fake", "model_commit_hash": commit}
        self.calls, self.rows = 0, 0
        self.fail_after_calls, self.nan, self.nondeterministic = fail_after_calls, nan, nondeterministic

    def _vec(self, t):
        seed = int(hashlib.sha1(t.encode()).hexdigest()[:8], 16)
        v = np.random.default_rng(seed).normal(size=DIM)
        return v / np.linalg.norm(v)

    def encode(self, texts):
        if self.fail_after_calls is not None and self.calls >= self.fail_after_calls:
            raise RuntimeError("simulated crash")
        self.calls += 1
        self.rows += len(texts)
        if self.nondeterministic:
            v = np.random.default_rng().normal(size=(len(texts), DIM))
            return (v / np.linalg.norm(v, axis=1, keepdims=True)).astype("float32")
        E = np.stack([self._vec(t) for t in texts]).astype("float32")
        if self.nan:
            E[0, 0] = np.nan
        return E


def _df(n=23, tokens=50):
    return pd.DataFrame({"chunk_id": [f"c{i}" for i in range(n)], "text": [f"chunk number {i}" for i in range(n)],
                         "n_tokens": [tokens] * n})


def _quiet(*a, **k):
    pass


def _run(df, emb, out, **kw):
    kw.setdefault("shard_size", 10)
    kw.setdefault("source_sha", "src1")
    return ec.embed_table(df, emb, Path(out), "T", log=_quiet, **kw)


def test_full_run_alignment_manifest_and_gates():
    with tempfile.TemporaryDirectory() as d:
        df, emb = _df(), FakeEmbedder()
        m = _run(df, emb, d)
        E = np.load(Path(d) / "T.npy")
        assert E.shape == (23, DIM) and E.dtype == np.float32
        fresh = FakeEmbedder()
        for i in (0, 9, 10, 22):  # across shard boundaries: row i must be the embedding of text i
            assert np.allclose(E[i], fresh.encode([df.text[i]])[0], atol=1e-6)
        assert m["rows"] == 23 and m["ids_sha256"] == ec.sha256_ids(df.chunk_id.tolist())
        assert m["query_prefix"] == ec.QUERY_PREFIX and m["passage_prefix"] == ""
        assert m["selfcheck_min_cosine"] > 0.9999
        assert json.loads((Path(d) / "T.manifest.json").read_text())["n_truncated_at_embedding"] == 0

# --------------------------------------------------------------- end-to-end synthetic world
def _world(seed=0, n_out=40, with_profile_advantage=True):
    """B has an extra 'profile' chunk per outlet very close to the outlet direction; A only has noisy window chunks.
    Queries sit near their outlet's direction -> B should find the gold outlet more often, by construction."""
    rng = np.random.default_rng(seed)
    keys = [f"{i:012x}" for i in range(n_out)]
    U = _unit(rng.normal(size=(n_out, D)))

    def build(name, per_outlet, profile):
        vecs, outlet, ctype, cid = [], [], [], []
        for i, k in enumerate(keys):
            if profile:
                vecs.append(_unit(U[i] + 0.25 * rng.normal(size=D)))
                outlet.append(k); ctype.append("profile"); cid.append(f"{k}:p")
            for j in range(per_outlet):
                vecs.append(_unit(U[i] + 1.1 * rng.normal(size=D)))
                outlet.append(k); ctype.append("review" if profile else "window"); cid.append(f"{k}:{j}")
        return ev.VariantData(name, np.array(vecs, dtype="float32"), np.array(outlet), np.array(ctype), np.array(cid))

    variants = {"A800": build("A800", 4, False), "B": build("B", 4, with_profile_advantage)}
    owners = [int(x) for x in rng.integers(0, n_out, 60)]
    gold = pd.DataFrame({"query_id": [f"q{i}" for i in range(60)], "query": [f"query text number {i} here" for i in range(60)],
                         "query_type": [ev.QUERY_TYPES[i % 3] for i in range(60)],
                         "gold_outlet_keys": [[keys[o]] for o in owners]})
    Q = _unit(np.array([U[o] + 0.5 * rng.normal(size=D) for o in owners])).astype("float32")
    return variants, Q, gold, set(keys)


def test_evaluate_end_to_end_direction_and_structure():
    variants, Q, gold, owr = _world()
    leaky = np.zeros(len(gold), bool)
    leaky[:3] = True
    exact_rows = {v: xs.exact_topk(d.E, Q, 150)[0] for v, d in variants.items()}
    report, pq = ev.evaluate(variants, Q, gold, leaky, owr, chroma_rows=exact_rows, n_boot=300)

    names = set(report["systems"])
    assert names == {"A800|exact", "A800|chroma", "B|exact", "B|chroma", "B_restricted|exact"}
    assert len(pq) == len(names) * len(gold)
    # B was built to be better: point estimate must favour it, and its 'profile' chunk must be what surfaces the gold outlet
    assert report["systems"]["B|exact"]["Hit@1"]["mean"] > report["systems"]["A800|exact"]["Hit@1"]["mean"]
    assert report["primary"]["hit5_diff"]["mean"] >= 0
    # bookkeeping check: the reported chunk type of the first gold hit must equal a direct recomputation
    hits = pq[(pq.system == "B|exact") & pq["rank"].notna()].set_index("query_id")
    v, ex = variants["B"], xs.exact_topk(variants["B"].E, Q, 150)[0]
    direct = []
    for q in range(len(gold)):
        rows = [r for r in ex[q] if v.outlet_of[r] == gold["gold_outlet_keys"].iat[q][0]]
        if rows:
            direct.append((gold["query_id"].iat[q], v.chunk_type[rows[0]]))
    assert {qid: ct for qid, ct in direct} == hits["first_gold_chunk_type"].to_dict()
    assert (hits["first_gold_chunk_type"] == "profile").mean() > 1 / 5  # chance level: 1 profile among 5 chunks
    # identical rows through the 'chroma' path -> engines agree exactly
    assert all(v["queries_where_hit5_differs"] == 0 for v in report["engine_agreement"].values())
    # structure of the report
    assert set(report["by_type"]) == set(ev.QUERY_TYPES)
    assert "B_vs_A800_excluding_leaky" in report["exploratory"] and report["exploratory"]["B_vs_A800_excluding_leaky"]["n"] == 57
    assert set(report["exploratory"]["depth_sensitivity"]) == {"top50_chunks", "top100_chunks"}
    assert report["primary"]["rr_wins_losses_ties"]["b_better"] + report["primary"]["rr_wins_losses_ties"]["a_better"] \
        + report["primary"]["rr_wins_losses_ties"]["tie"] == 60
    assert "inconclusive" in ev.verdict({"ci": [-1, 1]}) and isinstance(ev.format_summary(report), str)

def garble(s: str, times: int = 1) -> str:
    """Simulate the upstream bug: UTF-8 bytes decoded as cp1252 (optionally repeatedly)."""
    for _ in range(times):
        s = s.encode("utf-8").decode("cp1252", errors="replace")
    return s
